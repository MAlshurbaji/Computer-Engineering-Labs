# Draw a textured desk with printed detail; all returned arrays are RGB uint8.
def draw_scene(seed=1, kind='both'):
    generator = np.random.default_rng(seed)
    h, w = 128, 160
    yy, xx = np.mgrid[:h, :w]
    grain = 4 * np.sin(yy * 0.8 + xx * 0.04) + generator.normal(0, 1.2, (h, w))
    base = 145 + 22 * xx / w + grain
    rgb = np.stack([base + 18, base + 2, base - 15], axis=-1)
    canvas = Image.fromarray(np.clip(np.rint(rgb), 0, 255).astype('uint8'))
    draw = ImageDraw.Draw(canvas)
    if kind in ('both', 'book'):
        draw.rectangle((10, 17, 72, 98), fill=(35, 64, 105), outline=(15, 30, 52), width=2)
        draw.rectangle((10, 17, 18, 98), fill=(23, 42, 69))
        draw.text((24, 25), 'NOTES', fill=(240, 232, 201))
