# Drawn tabletop fixtures: RGB uint8, H=128, W=160, C=3.
def scene(kind="mug", seed=0):
    local = np.random.default_rng(seed)
    yy, xx = np.mgrid[:128, :160]
    grain = 4 * np.sin(xx / 3) + local.normal(0, 1, (128, 160))
    base = np.stack([205 + yy / 8 + grain, 191 + yy / 10 + grain,
                     166 + yy / 12 + grain], axis=-1)
    canvas = Image.fromarray(np.clip(base, 0, 255).astype(np.uint8))
    draw = ImageDraw.Draw(canvas)
    for y in range(12, 128, 24):
        draw.line((0, y, 159, y), fill=(184, 168, 146), width=1)
    if kind == "mug":
        draw.ellipse((105, 43, 139, 87), fill=(53, 118, 167))
        draw.ellipse((112, 50, 131, 80), fill=(208, 193, 168))
        draw.rounded_rectangle((43, 33, 115, 102), radius=12, fill=(66, 145, 194))
        draw.ellipse((43, 27, 115, 48), fill=(34, 71, 91))
        draw.ellipse((48, 31, 110, 43), fill=(116, 71, 40))
        draw.text((64, 64), "TEA", fill=(245, 246, 237))
    elif kind == "parcel":
        draw.rectangle((27, 31, 123, 101), fill=(174, 117, 65), outline=(70, 45, 28), width=2)
        draw.rectangle((69, 32, 80, 100), fill=(224, 190, 124))
        draw.rectangle((86, 48, 115, 77), fill=(247, 244, 225))
        draw.text((89, 55), "UP", fill=(25, 30, 34))
        draw.line((39, 64, 39, 45), fill=(25, 30, 34), width=3)
        draw.polygon([(33, 48), (45, 48), (39, 41)], fill=(25, 30, 34))
    elif kind == "apple":
        draw.ellipse((43, 38, 115, 108), fill=(190, 48, 40))
        draw.ellipse((58, 44, 75, 63), fill=(241, 123, 93))
        draw.line((80, 42, 86, 26), fill=(75, 63, 32), width=5)
        draw.ellipse((84, 25, 104, 35), fill=(54, 117, 56))
    else:
        raise ValueError("Unknown fixture kind")
    return np.asarray(canvas).copy()

def panels(images, titles, cmap=None):
    fig, axes = plt.subplots(1, len(images), figsize=(4 * len(images), 3.4), squeeze=False)
    for ax, image, title in zip(axes[0], images, titles):
        ax.imshow(image, cmap=cmap)
        ax.set_title(title)
        ax.axis("off")
    fig.tight_layout()
    return fig
