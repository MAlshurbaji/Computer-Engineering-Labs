# Original drawings represent two existing views of each of twelve household objects.
# Object groups, not individual views, are the independent units in this toy collection.
def draw_object(kind,index,view):
    height,width = 128,160
    yy,xx = np.mgrid[:height,:width]
    generator = np.random.default_rng(410+index+(100 if kind=="mug" else 0))
    grain = generator.normal(0,1.1,(height,width))+4*np.sin(yy*0.6)
    base = 174+18*xx/(width-1)+grain
    background = np.stack([base+16,base,base-22],axis=-1)
    image = Image.fromarray(np.clip(np.rint(background),0,255).astype(np.uint8))
    draw = ImageDraw.Draw(image)
    dx,dy = 3*view,-2*view
    if kind=="book":
        palette = [(166,50,58),(48,85,166),(58,116,77),(128,65,133),(151,100,35),(50,116,127)]
        color = palette[index-1]
        draw.rectangle((34+dx,13+dy,111+dx,110+dy),fill=color,outline=(35,37,43),width=2)
        draw.rectangle((35+dx,15+dy,44+dx,108+dy),fill=(46,44,48))
        draw.rectangle((52+dx,28+dy,100+dx,53+dy),fill=(240,231,205))
        draw.text((57+dx,34+dy),"NOTES",fill=(43,44,51))
        for y in range(91,107,3):
            draw.line((47+dx,y+dy,108+dx,y+dy),fill=(228,223,202),width=1)
    else:
        palette = [(232,229,210),(149,197,202),(230,188,136),(185,178,219),(194,214,162),(223,164,168)]
        color = palette[index-1]
        draw.ellipse((105+dx,49+dy,141+dx,91+dy),outline=color,width=8)
        draw.rounded_rectangle((45+dx,36+dy,112+dx,110+dy),radius=12,fill=color,outline=(86,89,91),width=2)
        draw.ellipse((45+dx,25+dy,112+dx,49+dy),fill=(240,235,216),outline=(79,77,65),width=2)
        draw.ellipse((51+dx,30+dy,106+dx,43+dy),fill=(86,49,28))
        draw.text((66+dx,66+dy),"TEA",fill=(50,79,93))
        draw.line((53+dx,49+dy,53+dx,95+dy),fill=(248,246,228),width=2)
    draw.rectangle((5,115,72,125),fill=(230,220,180))
    draw.text((9,115),f"item {index}",fill=(50,47,36))
    return np.asarray(image).copy()
