# Draw a textured desk with printed detail; all returned arrays are RGB uint8.
def draw_scene(seed=1, kind='both'):
    generator = np.random.default_rng(seed)
    h, w = 128, 160
    yy, xx = np.mgrid[:h, :w]
    grain = 4 * np.sin(yy * 0.8 + xx * 0.04) + generator.normal(0, 1.2, (h, w))
    base = 145 + 22 * xx / w + grain
    rgb = np.stack([base + 18, base + 2, base - 15], axis=-1)
    canvas = Image.fromarray(np.clip(np.rint(rgb), 0, 255).astype('uint8'))
    draw = ImageDraw.Draw(canvas)
    if kind in ('both', 'book'):
        draw.rectangle((10, 17, 72, 98), fill=(35, 64, 105), outline=(15, 30, 52), width=2)
        draw.rectangle((10, 17, 18, 98), fill=(23, 42, 69))
        draw.text((24, 25), 'NOTES', fill=(240, 232, 201))
        for y in range(49, 85, 5):
            draw.line((25, y, 62, y), fill=(94, 128, 153), width=1)
        for y in range(90, 97, 2):
            draw.line((20, y, 70, y), fill=(221, 217, 190), width=1)
    if kind in ('both', 'mug'):
        draw.ellipse((117, 48, 149, 84), outline=(235, 229, 205), width=6)
        draw.rounded_rectangle((89, 39, 127, 100), radius=7, fill=(223, 219, 199), outline=(116, 111, 97), width=2)
        draw.ellipse((89, 32, 127, 49), fill=(243, 235, 205), outline=(83, 73, 58), width=2)
        draw.ellipse((94, 36, 122, 45), fill=(73, 45, 29))
        draw.line((96, 55, 96, 90), fill=(247, 244, 227), width=2)
        draw.text((101, 68), 'TEA', fill=(76, 103, 111))
    draw.rectangle((77, 105, 152, 124), fill=(237, 215, 119), outline=(148, 119, 62))
    draw.text((82, 109), 'tea 4pm', fill=(50, 47, 34))
    return np.asarray(canvas).copy()
clean_rgb = draw_scene(1)
