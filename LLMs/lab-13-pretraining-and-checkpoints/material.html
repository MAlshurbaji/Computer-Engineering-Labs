Pretraining and Checkpoints

MaterialPracticeQuiz

Train a small notice-completion model and learn how to resume its work reliably. Follow the evidence from each update through a saved checkpoint and a final held-out evaluation.

What you will learn

  • Carry out bounded language-model training with real gradients.
  • Separate learning progress from held-out performance.
  • Save and restore model, optimizer, vocabulary and random state.
  • Select a checkpoint using validation data before one final test.

Python · shared setup
import math
import json
import tempfile
from pathlib import Path
import numpy as np
import torch
from torch import nn
from torch.nn import functional as F
import matplotlib.pyplot as plt
torch.set_num_threads(1)
torch.manual_seed(113)
rng = np.random.default_rng(113)
def as_array(tensor):
    return tensor.detach().cpu().numpy()
class CausalHeads(nn.Module):
    def __init__(self, width=16, heads=4, dropout=0.0):
        super().__init__()
        if width % heads:
            raise ValueError("width must be divisible by heads")
        self.width, self.heads, self.head_dim = width, heads, width // heads
        self.qkv = nn.Linear(width, 3*width, bias=False)
        self.proj = nn.Linear(width, width, bias=False)
        self.attn_drop = nn.Dropout(dropout)
    def forward(self, x):
        b,t,c = x.shape
        packed = self.qkv(x).reshape(b,t,3,self.heads,self.head_dim)
        q,k,v = packed.permute(2,0,3,1,4).unbind(0)
        scores = (q @ k.transpose(-2,-1)) / math.sqrt(self.head_dim)
        future = torch.triu(torch.ones(t,t,device=x.device,dtype=torch.bool),diagonal=1)
        weights = scores.masked_fill(future,float("-inf")).softmax(dim=-1)
        context = self.attn_drop(weights) @ v
        merged = context.transpose(1,2).contiguous().reshape(b,t,c)
        return self.proj(merged), weights
class FeedForward(nn.Module):
    def __init__(self,width=16,expansion=4):
        super().__init__()
        self.up = nn.Linear(width,expansion*width)
        self.act = nn.GELU()
        self.down = nn.Linear(expansion*width,width)
    def forward(self,x):
        return self.down(self.act(self.up(x)))
class TransformerBlock(nn.Module):
    def __init__(self,width=16,heads=4,dropout=0.0):
        super().__init__()
        self.norm1 = nn.LayerNorm(width)
        self.attention = CausalHeads(width,heads,dropout)
        self.norm2 = nn.LayerNorm(width)
        self.ff = FeedForward(width)
        self.drop = nn.Dropout(dropout)
    def forward(self,x):
        x = x + self.drop(self.attention(self.norm1(x))[0])
        return x + self.drop(self.ff(self.norm2(x)))
class TinyGPT(nn.Module):
    def __init__(self,vocab_size,context=12,width=16,heads=4,layers=2,dropout=0.0):
        super().__init__()
        self.config = dict(vocab_size=vocab_size,context=context,width=width,heads=heads,layers=layers,dropout=dropout)
        self.token = nn.Embedding(vocab_size,width)
        self.position = nn.Embedding(context,width)
        self.drop = nn.Dropout(dropout)
        self.blocks = nn.ModuleList([TransformerBlock(width,heads,dropout) for _ in range(layers)])
        self.final_norm = nn.LayerNorm(width)
    def forward(self,ids):
        if ids.ndim != 2 or ids.shape[1] == 0 or ids.shape[1] > self.config["context"]:
            raise ValueError("Expected nonempty B,T IDs within context")
        positions = torch.arange(ids.shape[1],device=ids.device)
        x = self.drop(self.token(ids) + self.position(positions)[None])
        for block in self.blocks:
            x = block(x)
        x = self.final_norm(x)
        return F.linear(x,self.token.weight)  # Tied input/output weights.
vocab = ["<pad>","<eos>","<unk>","library","opens","at","nine","ten","room","is","free","busy","please","return","the","book","today","tomorrow","desk","closes","six","seven","reserve","a","quiet","seat","now"]
stoi = {word:i for i,word in enumerate(vocab)}
def encode(text,eos=False):
    ids = [stoi.get(word,stoi["<unk>"]) for word in text.lower().split()]
    return ids + ([stoi["<eos>"]] if eos else [])
def decode(ids):
    return " ".join(vocab[int(i)] for i in ids)
train_documents = ["library opens at nine","please return the book today","room is free today","reserve a quiet seat","desk closes at six","room is busy now"]
validation_documents = ["library closes at seven","please return the book tomorrow"]
test_documents = ["room is free tomorrow","desk opens at ten"]
# Shift exactly once. Padding is outside each document's true target positions.
def make_batch(documents):
    sequences = [encode(text,eos=True) for text in documents]
    if not sequences or any(len(sequence)<2 for sequence in sequences):
        raise ValueError("Need nonempty documents with at least one word")
    lengths = torch.tensor([len(sequence) for sequence in sequences])
    padded = torch.full((len(sequences),int(lengths.max())),stoi["<pad>"],dtype=torch.long)
    for row,sequence in enumerate(sequences):
        padded[row,:len(sequence)] = torch.tensor(sequence)
    inputs = padded[:,:-1].clone()
    targets = padded[:,1:].clone()
    valid = torch.arange(targets.shape[1])[None] < (lengths-1)[:,None]
    targets[~valid] = -100
    return inputs,targets

def token_loss(model,inputs,targets):
    logits = model(inputs)
    return F.cross_entropy(logits.reshape(-1,len(vocab)),targets.reshape(-1),ignore_index=-100)
def update(model,optimizer,inputs,targets):
    model.train()
    optimizer.zero_grad(set_to_none=True)
    loss = token_loss(model,inputs,targets)
    loss.backward()
    optimizer.step()
    return float(loss.detach())
def measure(model,documents):
    previous = model.training
    model.eval()
    loss_sum,count = 0.0,0
    try:
        with torch.no_grad():
            for text in documents:
                inputs,targets = make_batch([text])
                loss_sum += float(F.cross_entropy(model(inputs).reshape(-1,len(vocab)),targets.reshape(-1),ignore_index=-100,reduction="sum"))
                count += int((targets!=-100).sum())
    finally:
        model.train(previous)
    return loss_sum/count

Guided task 01

One optimization update

Pretraining repeatedly asks a model to predict later tokens from earlier ones. Our tiny notice collection supplies the target tokens automatically; no person labels each next word. Here pretraining means learning from randomly initialized parameters on those notices, not downloading an already capable language model. The fixed vocabulary and document split come from setup.

An update has four distinct stages: clear old gradients, compute a scalar loss, differentiate it, then ask the optimizer to change parameters. PyTorch accumulates gradients by default, so skipping the first stage changes the meaning of repeated updates. AdamW also keeps internal statistics; its state will matter when we resume training. Start with one full batch so data ordering does not obscure these operations.

Try it

  1. Create a width-8 model with one transformer block and no dropout.
  2. Use AdamW with learning rate 0.01 and zero weight decay for this demonstration.
  3. Compare the tied token weights before and after one update.
Python · run in order
inputs,targets = make_batch(train_documents)
learner = TinyGPT(len(vocab),width=8,heads=2,layers=1)
optimizer = torch.optim.AdamW(learner.parameters(),lr=0.01,weight_decay=0.0)
before = learner.token.weight.detach().clone()
first_loss = update(learner,optimizer,inputs,targets)
changes = (learner.token.weight.detach()-before).norm(dim=1)
print("Training loss before this update:",first_loss)
fig,ax = plt.subplots(figsize=(7,3.4))
ax.bar(vocab,as_array(changes))
ax.set(title="Measured embedding changes",ylabel="L2 change")
fig.tight_layout()
ax.tick_params(axis="x",rotation=75)

Read the result

Every bar measures a real parameter change from this update. Some vocabulary entries can change even when absent from inputs because the same embedding matrix also produces all output logits. The loss was measured before optimizer.step, so it does not describe the updated model.

Change and compare

  • Set the learning rate to 0 and confirm that the measured changes disappear.
  • Inspect len(optimizer.state) after the update; relate those entries to later checkpoint storage.

Guided task 02

Bounded training and measured curves

A short loop makes repeated updates visible without hiding them inside a training framework. Continue the same model for twenty-five updates. At each fifth update, measure both training and validation loss with dropout disabled and gradients off. These measurements differ from the loss returned inside an update: they use the current parameters after that update and the same evaluation procedure for both document sets.

The validation notices are not passed to backward or optimizer.step. Their wording overlaps the teaching vocabulary but their complete documents remain separate. With only a few examples, the curve is a diagnostic trace, not a convincing estimate of performance on new campus messages. A falling training curve shows fitting; the validation curve asks whether that fitting helps on these held-out examples.

Try it

  1. Record a step-zero measurement for the current model.
  2. Perform 25 more full-batch updates, recording every fifth step.
  3. Plot the measured losses against completed updates.
Python · run in order
steps = [0]
training_curve = [measure(learner,train_documents)]
validation_curve = [measure(learner,validation_documents)]
for step in range(1,26):
    update(learner,optimizer,inputs,targets)
    if step%5==0:
        steps.append(step)
        training_curve.append(measure(learner,train_documents))
        validation_curve.append(measure(learner,validation_documents))
fig,ax = plt.subplots(figsize=(6.6,3.7))
ax.plot(steps,training_curve,"o-",label="Training notices")
ax.plot(steps,validation_curve,"s-",label="Validation notices")

ax.set_xticks(steps)
ax.legend()
fig.tight_layout()

Read the result

Compare the two trajectories without assuming that either must decrease at every measurement. The validation loss uses every intended document and weights supervised tokens equally. No result from the final test set has influenced this trace.

Guided task 03

Gradient norms and clipping

Gradient clipping limits the size of a gradient vector before the optimizer uses it. It does not cap the loss, directly clamp model parameters, or guarantee that a learning rate is appropriate. To make the effect easy to inspect, multiply a fresh model's loss by fifty in this diagnostic. That artificial scaling is deliberate; it is not a recommended training objective.

The global L2 norm treats all parameter gradients as one long vector. clip_grad_norm_ returns the norm before clipping and modifies gradients in place. We then recompute the norm to check what changed. This task stops before an optimizer update, keeping the evidence focused on gradients. In a real loop, clipping belongs after backward and before optimizer.step.

Try it

  1. Create a fresh diagnostic model and differentiate fifty times its token loss.
  2. Clip its global gradient norm to 0.5.
  3. Compare the returned original norm with the recomputed clipped norm.
Python · run in order
probe = TinyGPT(len(vocab),width=8,heads=2,layers=1)
probe.zero_grad(set_to_none=True)
scaled_loss = 50*token_loss(probe,inputs,targets)
scaled_loss.backward()
original_norm = float(torch.nn.utils.clip_grad_norm_(probe.parameters(),max_norm=0.5))
clipped_norm = float(torch.sqrt(sum(p.grad.detach().square().sum() for p in probe.parameters() if p.grad is not None)))
fig,ax = plt.subplots(figsize=(7,3.4))
ax.bar(['Before clipping','After clipping'],[original_norm,clipped_norm])
ax.set(title="Measured gradient norms",ylabel="Global L2 norm")
fig.tight_layout()

Read the result

The bar after clipping should meet the chosen norm bound within floating-point tolerance. The original norm can be much larger because this experiment deliberately scaled the loss. Clipping alone says nothing about generalization, and it is not a substitute for investigating persistent unstable gradients.

Change and compare

  • Use a maximum norm larger than the original norm and observe whether clipping changes gradients.
  • Remove the factor of 50 and compare magnitudes without assuming the same clipping frequency.

Guided task 04

Independent parameter snapshots

A checkpoint should represent a particular moment, even if training continues. A model's state_dict is a mapping whose tensors can still share storage with live parameters. Merely assigning that mapping to another variable therefore does not create an independent best-model snapshot. This subtle aliasing can silently replace yesterday's best values with today's later values.

For an in-memory snapshot, detach and clone every tensor. detach removes its connection to autograd; clone creates separate storage. Serializing immediately to a file also records the current values, but a dictionary kept in memory needs this explicit care. Use one more update to test the difference. Comparing values before and after provides stronger evidence than relying on a variable named best.

Try it

  1. Keep one live state dictionary and one dictionary of detached clones.
  2. Perform one additional training update.
  3. Measure whether each stored token-weight tensor changed.
Python · run in order
live_state = learner.state_dict()
frozen_state = {name:value.detach().clone() for name,value in learner.state_dict().items()}
snapshot_token = frozen_state["token.weight"].clone()
update(learner,optimizer,inputs,targets)
live_change = float((live_state["token.weight"]-snapshot_token).abs().max())
frozen_change = float((frozen_state["token.weight"]-snapshot_token).abs().max())
fig,ax = plt.subplots(figsize=(7,3.4))
ax.bar(['Live mapping','Cloned snapshot'],[live_change,frozen_change])
ax.set(title="Snapshot storage test",ylabel="Maximum absolute change")
fig.tight_layout()

Read the result

The live mapping follows the update while the cloned snapshot stays fixed. This does not mean state_dict is faulty: it is designed to expose module state. The important choice is whether your workflow needs a live view, a separate copy, or a serialized record.

Change and compare

  • Check data_ptr for the live and cloned token tensors to compare storage identities.
  • Load the frozen state into a separate matching model and verify that it differs from the updated learner.

Guided task 05

A complete checkpoint record

Weights alone are enough for some inference uses, but not for reconstructing a training run. A saved record should identify the architecture, token ordering, preprocessing convention and completed progress. AdamW's internal statistics are also necessary to continue its trajectory. Recording only a seed is insufficient once random draws have already occurred; record the current random-generator state when exact continuation matters.

This checkpoint contains tensors and simple Python containers that weights_only loading can handle. It is saved in a newly created temporary directory, so rerunning the task does not overwrite a previous run. Our CPU loop uses a fixed full batch and no random data ordering. The PyTorch CPU random state therefore covers the only random stream needed for its continuation; a shuffled loader or other random augmentation would require its own state too.

Try it

  1. Save the current model, optimizer, configuration and fixed vocabulary.
  2. Record 27 completed updates and the CPU random state.
  3. Print the temporary path and inspect the major tensor counts.

Guided task 06

Reloading for equivalent predictions

Loading reconstructs a model from its saved configuration and then fills its parameters. It does not infer architecture from a filename. Check the vocabulary ordering and tokenizer convention before feeding text: exchanging two token IDs can preserve every tensor shape while changing the meaning of predictions. Strict state loading helps detect missing parameters and shape mismatches, but it cannot detect a semantically reordered vocabulary.

For this task, compare an evaluation-mode forward pass before and after restoration. No training step occurs, and no sampled text is involved, so output equality is a direct test of the saved model. weights_only limits the objects accepted by loading; it is still sensible to use files you created or obtain from a trusted source. The example reads only its own temporary checkpoint.

Try it

  1. Load the local checkpoint onto the CPU with weights_only=True.
  2. Build a new model from its configuration and verify its token mapping.
  3. Compare logits on a short room-booking prefix.
Python · run in order
loaded = torch.load(checkpoint_path,map_location="cpu",weights_only=True)
if loaded["vocab"]!=vocab or loaded["tokenizer"]!="lowercase-whitespace-v1":
    raise ValueError("Incompatible tokenizer")
restored = TinyGPT(**loaded["config"])
restored.load_state_dict(loaded["model"],strict=True)
restored.eval()
learner.eval()
prefix = torch.tensor([encode("room is")])
with torch.no_grad():
    expected_logits = learner(prefix)
    restored_logits = restored(prefix)
difference = (expected_logits-restored_logits).abs()[0]
fig,ax = plt.subplots(figsize=(7,3.3))
artist = ax.imshow(as_array(difference),vmin=0,vmax=max(float(difference.max()),1e-8),aspect="auto",cmap="viridis")
ax.set(title="Reloaded logit differences",xlabel="Vocabulary ID",ylabel="Prefix position")

fig.tight_layout()

Read the result

The zero difference tests the serialization round trip for this configuration and prefix. A successful reload does not demonstrate model quality. It demonstrates that the stored artifact reproduces these numerical predictions, which is a prerequisite for meaningful later comparisons.

Guided task 07

Exact continuation with random state

An exact-resume test asks a stricter question than inference equality: does the next update match an uninterrupted run? This experiment enables dropout, trains one step, saves state, and branches. The uninterrupted branch takes its next update immediately. The restored branch rebuilds the model and optimizer, reloads both, then restores the CPU random state immediately before taking that same update.

Model construction consumes random numbers, so restoring the generator before construction would put it in the wrong position by the time dropout runs. The data batch and update procedure must also match. This small CPU experiment has fixed ordering and one active random stream; exact results across different hardware, library versions or nondeterministic operations are not promised. Record the environment when transporting a real experiment.

Try it

  1. Save after one update of a fresh model with dropout 0.2.
  2. Run one uninterrupted update and retain its resulting parameters.
  3. Rebuild, reload and restore RNG last; compare the next update.
Python · run in order
torch.manual_seed(1313)
branch = TinyGPT(len(vocab),width=8,heads=2,layers=1,dropout=0.2)
branch_optimizer = torch.optim.AdamW(branch.parameters(),lr=0.005,weight_decay=0.0)
update(branch,branch_optimizer,inputs,targets)
resume_path = scratch/"resume.pt"
torch.save({"config":branch.config,"model":branch.state_dict(),"optimizer":branch_optimizer.state_dict(),"torch_rng":torch.get_rng_state(),"vocab":list(vocab),"tokenizer":"lowercase-whitespace-v1","completed_steps":1,"data_order":"fixed-full-batch-v1"},resume_path)
continuous_loss = update(branch,branch_optimizer,inputs,targets)
continuous_state = {name:value.detach().clone() for name,value in branch.state_dict().items()}
resume = torch.load(resume_path,weights_only=True,map_location="cpu")
restarted = TinyGPT(**resume["config"])
restarted.load_state_dict(resume["model"])
restarted_optimizer = torch.optim.AdamW(restarted.parameters(),lr=0.005,weight_decay=0.0)
restarted_optimizer.load_state_dict(resume["optimizer"])
torch.set_rng_state(resume["torch_rng"])
resumed_loss = update(restarted,restarted_optimizer,inputs,targets)
parameter_errors = [float((continuous_state[name]-value).abs().max()) for name,value in restarted.state_dict().items()]
fig,ax = plt.subplots(figsize=(7,3.4))
ax.bar(['Loss difference','Largest parameter difference'],[abs(continuous_loss-resumed_loss),max(parameter_errors)])
ax.set(title="Exact next-update comparison",ylabel="Absolute difference")
ax.set_ylim(0,1)
ax.text(0.5,0.5,"Both differences are zero" if max(parameter_errors)==0 and continuous_loss==resumed_loss else "Inspect the measured differences",ha="center",transform=ax.transAxes)
fig.tight_layout()

Read the result

The two paths should produce identical loss and parameters in this controlled CPU run. Matching only the starting seed would restart a random sequence, not continue it. An optimizer freshly initialized at the saved weights would also omit the running statistics accumulated by the earlier update.

  • Omit the optimizer reload in a separate trial and compare the resulting parameter difference.
  • Omit the RNG restoration while retaining dropout, then explain which part of the computation can change.

Guided task 08

Validation selection and one final test

Checkpoint selection and checkpoint resumption solve different problems. The latest checkpoint supports continuing a run; the checkpoint with the best validation measurement is a candidate for deployment or final evaluation. Run a fresh short experiment and retain independent states after four fixed training intervals. Select the lowest validation loss, with the earlier candidate winning a tie.

Only after selection do we evaluate the chosen state on the final test notices. The test result is recorded once; it does not decide which checkpoint wins or suggest another training duration. Our two test documents are far too small for a reliable application claim, but the information flow is correct. More realistic work needs larger grouped splits, uncertainty analysis and a clearly stated target population.

Try it

  1. Train a fresh model for four intervals of five updates.
  2. Clone each candidate and choose by validation loss alone.
  3. Restore the winner and measure the final test once.
Python · run in order
candidate = TinyGPT(len(vocab),width=8,heads=2,layers=1)
candidate_optimizer = torch.optim.AdamW(candidate.parameters(),lr=0.01,weight_decay=0.0)
candidate_steps,candidate_losses,candidate_states = [],[],[]
for interval in range(4):
    for unused in range(5):
        update(candidate,candidate_optimizer,inputs,targets)
    candidate_steps.append((interval+1)*5)
    candidate_losses.append(measure(candidate,validation_documents))
    candidate_states.append({name:value.detach().clone() for name,value in candidate.state_dict().items()})
best_index = min(range(len(candidate_losses)),key=lambda i:candidate_losses[i])
candidate.load_state_dict(candidate_states[best_index])
final_test_loss = measure(candidate,test_documents)
print("Selected step:",candidate_steps[best_index],"Final test loss:",final_test_loss)
fig,ax = plt.subplots(figsize=(6.4,3.6))
ax.plot(candidate_steps,candidate_losses,"o-",label="Validation")
ax.scatter([candidate_steps[best_index]],[candidate_losses[best_index]],s=100,facecolors="none",edgecolors="red",label="Selected")
ax.set(xlabel="Completed updates",ylabel="Mean token loss",title="Validation chooses the checkpoint")
ax.set_xticks(candidate_steps)
ax.legend()
fig.tight_layout()

Read the result

The circled checkpoint follows the declared validation rule, even if a later checkpoint has lower training loss. The printed test loss measures the chosen artifact rather than participating in its selection. Saving the selected configuration and vocabulary together would make that artifact reconstructible.

Change and compare

  • Use a fresh development run to compare a different learning rate without consulting the saved final test result.
  • Describe why choosing the minimum test loss across these four checkpoints would change the role of the test set.

Extension 1

Optional: warmup and cosine decay

Appendix D adds useful training refinements. A learning-rate schedule specifies the rate used at each update; gradient clipping separately controls unusually large gradients. Neither guarantees improved validation performance. Define the schedule and update budget before comparing runs, and keep the data, initialization and other settings fixed for a controlled comparison.

This standalone calculator uses zero-based update indices. The first warmup update uses peak/warmup, the last warmup update reaches peak, and the final planned update reaches minimum. Use lr_at(completed_steps, ...) to set each optimizer parameter group’s learning rate immediately before its next update. When resuming, restore the completed-step count and these schedule settings with the optimizer state; restarting the warmup would define a different run. The printed values describe a schedule, not measured training performance.

Python · standalone extension
import math

def lr_at(step, *, warmup, total, peak, minimum):
    if not (isinstance(step,int) and isinstance(warmup,int) and isinstance(total,int)):
        raise ValueError('Step counts must be integers.')
    if not (0 <= step < total and 1 <= warmup < total):
        raise ValueError('Require a valid update index and 1 <= warmup < total.')
    if not (math.isfinite(peak) and math.isfinite(minimum) and 0 <= minimum <= peak):
        raise ValueError('Require finite rates with 0 <= minimum <= peak.')
    if step < warmup:
        return peak * (step + 1) / warmup
    fraction = (step - warmup + 1) / (total - warmup)
    return minimum + 0.5 * (peak - minimum) * (1 + math.cos(math.pi * fraction))

settings = dict(warmup=4,total=20,peak=0.001,minimum=0.0001)
rates = [lr_at(step,**settings) for step in range(settings['total'])]
for step,rate in enumerate(rates):
    print(f'Update {step+1:02}: learning rate {rate:.7f}')
assert math.isclose(rates[3],settings['peak'])
assert math.isclose(rates[-1],settings['minimum'])
assert all(a >= b for a,b in zip(rates[3:-1],rates[4:]))
