Instruction Fine-Tuning

MaterialPracticeQuiz

Train a small model with a precisely defined instruction objective. You will compare held-out behavior before and after the update, then preserve enough information to replay the result.

What you will learn

  • Train only on response and real EOS targets.
  • Account for shared parameters and supervised tokens.
  • Compare loss and generated text on the same held-out requests.
  • Save a model with its formatting and vocabulary contract.

Before this lab: Use Python, token IDs, tensor axes and the causal-transformer concepts from preceding labs.

In this lab

  1. Establish the held-out baseline
  2. Inspect the supervised training objective
  3. Declare parameter sharing and the update budget
  4. Run bounded response-only updates
  5. Verify token-weighted held-out loss
  6. Compare free responses on the same prompts
  7. Inspect termination separately from fluent text
  8. Preserve an instruction checkpoint and replay it

import copy, json, re, tempfile
from pathlib import Path
import numpy as np
import torch
from torch import nn
from torch.nn import functional as F
import matplotlib.pyplot as plt
torch.set_num_threads(1)
torch.manual_seed(119)
# Fixed character vocabulary: never fitted on held-out responses.
characters = '\n' + ''.join(chr(i) for i in range(32,127))
char_to_id = {character:i+3 for i,character in enumerate(characters)}
id_to_char = {value:key for key,value in char_to_id.items()}
BOS, EOS, PAD = 1, 2, 2
VOCAB_SIZE = len(char_to_id)+3
instruction_rows = [
    {'id':'train0','split':'train','input':'library','response':'bring a book'},
    {'id':'train1','split':'train','input':'park','response':'bring water'},
    {'id':'train2','split':'train','input':'rain','response':'bring an umbrella'},
    {'id':'train3','split':'train','input':'gym','response':'bring a towel'},
    {'id':'train4','split':'train','input':'school','response':'bring a book'},
    {'id':'train5','split':'train','input':'picnic','response':'bring water'},
    {'id':'train6','split':'train','input':'showers','response':'bring an umbrella'},
    {'id':'train7','split':'train','input':'swimming','response':'bring a towel'},
    {'id':'val0','split':'validation','input':'reading cafe','response':'bring a book'},
    {'id':'val1','split':'validation','input':'workout','response':'bring a towel'},
    {'id':'test0','split':'test','input':'hot day','response':'bring water'},
    {'id':'test1','split':'test','input':'wet weather','response':'bring an umbrella'},
]
for row in instruction_rows:
    row['instruction'] = 'name one useful item'
    row['group'] = row['id']  # Each supplied situation has one record.

def encode_text(text):
    if any(character not in char_to_id for character in text):
        raise ValueError('Only the declared character vocabulary is supported.')
    return [char_to_id[character] for character in text]

def format_prompt(row):
    return 'Task: '+row['instruction']+'\nInput: '+row['input']+'\nReply: '

def instruction_parts(row):
    response = row['response'].strip()
    if not response:
        raise ValueError('A response must contain visible text.')
    prefix = [BOS]+encode_text(format_prompt(row))
    target = encode_text(response)+[EOS]
    return prefix,target

def pack_instructions(rows,max_length=128):
    if not rows:
        raise ValueError('Cannot collate an empty batch.')
    parts = [instruction_parts(row) for row in rows]
    lengths = [len(prefix)+len(target) for prefix,target in parts]
    if any(length>max_length for length in lengths):
        raise ValueError('Complete response and EOS must fit; shorten or reject the record.')
    ids = torch.full((len(rows),max(lengths)),PAD,dtype=torch.long)
    valid = torch.zeros_like(ids,dtype=torch.bool)
    labels = torch.full_like(ids,-100)
    for i,(prefix,target) in enumerate(parts):
        sequence = prefix+target
        ids[i,:len(sequence)] = torch.tensor(sequence)
        valid[i,:len(sequence)] = True
        labels[i,len(prefix):len(sequence)] = torch.tensor(target)
    return ids,valid,labels
# One small causal transformer block, randomly initialized: not pretrained.
class TinyBackbone(nn.Module):
    def __init__(self, vocabulary_size, width=24, context=32):
        super().__init__()
        self.token = nn.Embedding(vocabulary_size, width)
        self.position = nn.Embedding(context, width)
        self.norm1 = nn.LayerNorm(width)
        self.attention = nn.MultiheadAttention(width, 4, dropout=0.1, batch_first=True)
        self.norm2 = nn.LayerNorm(width)
        self.ff = nn.Sequential(nn.Linear(width, width*2), nn.GELU(), nn.Linear(width*2, width))
        self.final_norm = nn.LayerNorm(width)
    def forward(self, ids, valid):
        if ids.ndim != 2 or ids.shape != valid.shape or ids.shape[1] > self.position.num_embeddings:
            raise ValueError('Invalid token shape or context length.')
        if not torch.all(valid.any(dim=1)):
            raise ValueError('Every row needs a real token.')
        length = ids.shape[1]
        x = self.token(ids) + self.position(torch.arange(length, device=ids.device))[None]
        causal = torch.triu(torch.ones(length, length, dtype=torch.bool, device=ids.device), diagonal=1)
        normalized = self.norm1(x)
        attended, _ = self.attention(normalized, normalized, normalized, attn_mask=causal, key_padding_mask=~valid, need_weights=False)
        x = x + attended
        return self.final_norm(x + self.ff(self.norm2(x)))

class TinyInstructionLM(nn.Module):
    def __init__(self):
        super().__init__()
        self.backbone = TinyBackbone(VOCAB_SIZE,width=24,context=128)
        self.output = nn.Linear(24,VOCAB_SIZE,bias=False)
        self.output.weight = self.backbone.token.weight
    def forward(self,ids,valid):
        return self.output(self.backbone(ids,valid))

def response_loss_sum(logits,labels):
    targets = labels[:,1:]  # The local model does not shift inside forward.
    count = int((targets!=-100).sum())
    if count==0:
        raise ValueError('No supervised next-token targets.')
    total = F.cross_entropy(logits[:,:-1].reshape(-1,VOCAB_SIZE),targets.reshape(-1),ignore_index=-100,reduction='sum')
    return total,count

def evaluate_instructions(model,rows,batch_size=2):
    if not rows or batch_size<1:
        raise ValueError('Need records and a positive batch size.')
    model.eval()
    total,count = 0.0,0
    with torch.no_grad():
        for start in range(0,len(rows),batch_size):
            ids,valid,labels = pack_instructions(rows[start:start+batch_size])
            loss,n = response_loss_sum(model(ids,valid),labels)
            total += float(loss)
            count += n
    return {'loss':total/count,'tokens':count,'records':len(rows)}

def generate_reply(model,row,max_new_tokens=24):
    if max_new_tokens<1:
        raise ValueError('Generation budget must be positive.')
    sequence = [BOS]+encode_text(format_prompt(row))
    if len(sequence)>=128:
        raise ValueError('The prompt leaves no generation space.')
    generated = []
    reason = 'budget'
    model.eval()
    with torch.no_grad():
        for _ in range(max_new_tokens):
            if len(sequence)>=128:
                reason='context'
                break
            ids = torch.tensor([sequence])
            scores = model(ids,torch.ones_like(ids,dtype=torch.bool))[0,-1].clone()
            scores[0] = scores[BOS] = -torch.inf
            next_id = int(scores.argmax())
            sequence.append(next_id)
            generated.append(next_id)
            if next_id==EOS:
                reason='eos'
                break
    text = ''.join(id_to_char[token] for token in generated if token in id_to_char)
    return {'text':text,'token_ids':generated,'stop':reason}
# Small plotting helpers display measured tensors and counts.
def bar_plot(labels, values, title, ylabel):
    fig, ax = plt.subplots(figsize=(7, 3.5))
    ax.bar(labels, np.asarray(values))
    ax.set(title=title, ylabel=ylabel)
    if np.all(np.asarray(values)==0):
        ax.set_ylim(0,1)
        ax.text(0.5,0.5,'All values are zero',transform=ax.transAxes,ha='center')
    fig.tight_layout()
    return fig

def heat_plot(values, title, xlabel, ylabel, label):
    array = values.detach().cpu().numpy() if isinstance(values, torch.Tensor) else np.asarray(values)
    fig, ax = plt.subplots(figsize=(7, 3.5))
    if np.any(array < 0):
        bound = max(float(np.abs(array).max()), 1e-8)
        view = ax.imshow(array, aspect='auto', interpolation='nearest', cmap='coolwarm', vmin=-bound, vmax=bound)
    else:
        view = ax.imshow(array, aspect='auto', interpolation='nearest', cmap='viridis', vmin=0, vmax=max(float(array.max()),1e-8))
    ax.set_xticks(np.unique(np.linspace(0,array.shape[1]-1,min(8,array.shape[1])).astype(int)))
    ax.set_yticks(np.unique(np.linspace(0,array.shape[0]-1,min(8,array.shape[0])).astype(int)))
    ax.set(title=title, xlabel=xlabel, ylabel=ylabel)
    fig.colorbar(view, ax=ax, label=label)
    fig.tight_layout()
    return fig

Guided task 01

Establish the held-out baseline

The core model begins with random parameters; it has not inherited the language knowledge of a pretrained model. Its single causal block, small feature width and character vocabulary make the training mechanism inexpensive to inspect. Calling this run instruction training describes its supervision format, not evidence that it already understands requests. The optional extension later starts from genuine GPT-2 weights and keeps that checkpoint architecture intact.

First select training and validation records using the fixed roles in setup. Create a baseline model and copy it into a candidate so both start with exactly the same parameters. Measure validation response loss before any update. This scalar measures teacher-forced next-token prediction on the declared response characters and EOS. It does not measure the quality of a freely generated answer. Keeping the original baseline object unchanged also makes paired generation possible after training. Do not replace it with a newly initialized model: a different random starting point would confound the comparison you are trying to make.

Try it

  1. Collect training and validation records without opening test outcomes.
  2. Clone one initialized model to create the candidate.
  3. Measure the untouched validation objective.

Guided task 02

Inspect the supervised training objective

The collator provides input IDs, attention validity and unshifted response labels. The model consumes only IDs and validity. response_loss_sum aligns local logits with the next-position labels once, ignores the prompt and padding labels, and returns both the summed cross entropy and the number of supervised targets. Dividing those two quantities defines the mean response loss used for an update.

EOS contributes one real target per training record. That target teaches when to stop, while the remaining targets teach the response characters. Count both categories independently before training. If a formatter change shifts the response boundary, these counts and the earlier mask invariants help locate the problem. The attention mask still permits response positions to use the request as context. The response objective is therefore conditional on the prompt even though the prompt itself is not directly scored. In a larger collection, record lengths vary more; using a token denominator makes the weighting policy explicit instead of accidentally assigning equal influence to differently sized batches.

Try it

  1. Collate the fixed training records and inspect label counts.
  2. Separate visible response targets from genuine EOS targets.
  3. Calculate a finite mean loss with the stated denominator.
Python · run in order
train_ids,train_valid,train_labels=pack_instructions(train_rows)

Guided task 03

Declare parameter sharing and the update budget

The output projection reuses the token embedding weight. Tying these weights means one parameter tensor serves two computational roles. It does not mean two independent copies should be passed to the optimizer. PyTorch model.parameters enumerates shared parameters once, which keeps parameter accounting and optimizer state consistent with the actual model.

For this small core experiment, every unique parameter is trainable. The budget is fixed at forty full-batch updates with AdamW, learning rate 0.003 and weight decay 0.01. These are supplied demonstration settings, not values selected by examining the final test set. The later real pretrained extension uses a narrower policy: only the final transformer block and final normalization are adapted, keeping the larger run manageable. These policies answer different practical constraints but both must be stated explicitly. Save the training configuration with any result. A lower loss after more work is not directly comparable with a shorter run unless the difference in update budget is visible.

Try it

  1. Verify that input and output weights refer to the same parameter.
  2. Count unique trainable parameters without duplicating the tied tensor.
  3. Create the optimizer and fixed forty-step configuration.
Python · run in order
config={'steps':40,'learning_rate':0.003,'weight_decay':0.01,'context':128,'width':24,'heads':4,'seed':119}

Guided task 04

Run bounded response-only updates

Each update starts by enabling training mode and clearing previous gradients. The forward pass computes logits under the causal and padding contracts; the response loss then selects the supervised next-token targets. Backward computes gradients and the optimizer changes the declared parameters. The model uses attention dropout during training, so evaluation must explicitly switch to eval mode rather than merely disabling gradient recording.

The loop records the training objective every step and evaluates the fixed validation requests every ten steps. evaluate_instructions leaves the model in evaluation mode, so the next update calls train again. This small detail prevents an evaluation call from silently changing subsequent training behavior. The plot reports actual observations from this run. The two curves need not move together, and a falling training curve does not prove better general instruction following. Keep the baseline untouched and preserve the predetermined stopping budget even if a tiny held-out score fluctuates. The objective here is an inspectable optimization experiment with honest limits.

Try it

  1. Perform forty updates using training records only.
  2. Evaluate all validation records after each ten-step interval.
  3. Plot measured training and validation losses with their respective positions.
Python · run in order
training_curve=[]
validation_steps=[0]
validation_curve=[before['loss']]
for step in range(1,config['steps']+1):
    candidate.train()
    optimizer.zero_grad(set_to_none=True)
    loss_sum,n=response_loss_sum(candidate(train_ids,train_valid),train_labels)
    loss=loss_sum/n
    loss.backward()
    optimizer.step()
    training_curve.append(float(loss.detach()))
    if step%10==0:
        validation_steps.append(step)
        validation_curve.append(evaluate_instructions(candidate,val_rows)['loss'])
fig,ax=plt.subplots(figsize=(7,3.5))
ax.plot(range(1,41),training_curve,label='Training response loss')
ax.plot(validation_steps,validation_curve,'o-',label='Validation response loss')
ax.set(xlabel='Optimizer update',ylabel='Mean loss per supervised token',title='Measured bounded instruction training')
ax.legend()
fig.tight_layout()

Read the result

Guided task 05

Verify token-weighted held-out loss

Instruction responses can have unequal token counts. If one batch contains a short answer and another a longer answer, averaging their mean losses equally gives each batch the same weight rather than each supervised token. The evaluation helper instead adds summed target losses and divides by the total number of supervised targets across every intended record.

Recompute the same validation score with batch sizes one and two. The result should agree within floating-point tolerance because batching changes storage layout, not the objective definition. The code includes all records and the real EOS targets. It runs in evaluation mode with no gradient graph, which removes attention dropout and unnecessary gradient storage. The before/after bars use the same requests, vocabulary, formatter and denominator. A change in the score therefore describes the candidate relative to its unchanged starting state on this exact fixture. It remains a narrow teacher-forced measurement, not an estimate of broad task success or a replacement for inspecting generated replies.

Try it

  1. Evaluate the complete validation set with two different batch sizes.
  2. Check total target counts and numerical agreement.
  3. Plot the paired baseline and candidate objective.

after_two=evaluate_instructions(candidate,val_rows,batch_size=2)

Guided task 06

Compare free responses on the same prompts

Teacher forcing supplies the correct earlier response characters. Free generation supplies the model with its own previous predictions instead, so one early mistake can alter the remaining context. These are related but different experiments. Evaluate free responses separately with prompt-only inputs, identical greedy decoding and a fixed maximum of twenty-four new tokens.

The helper excludes reserved zero and BOS from generated choices, stops on EOS, and records whether EOS, the generation budget or the context limit ended the continuation. It returns only newly generated token IDs and their decoded characters. The initial prompt is never part of the returned response text. Preserve the actual strings rather than replacing awkward output with a cleaner illustration. A tiny randomly initialized model may repeat characters, emit an empty answer or produce a familiar fragment without selecting the correct item. Those failures are useful evidence about the limits of a short training run. The optional pretrained route follows the same paired comparison principle with its own tokenizer.

Guided task 07

Inspect termination separately from fluent text

A termination token is a learned prediction, not a timer. The generation helper also enforces external limits so an uncooperative model cannot run indefinitely. These two mechanisms should be recorded separately: reaching EOS indicates the model selected termination; reaching the budget indicates the application stopped a continuation that might otherwise continue.

Measure the probability assigned to the genuine EOS target under teacher-forced validation context. This uses the last visible reference character as the prediction position. It is a diagnostic of termination under correct earlier text, not the probability that free generation will stop successfully after an arbitrary path. Compare it with the actual stopping reasons already recorded. A model can assign a reasonable EOS probability at the reference endpoint yet wander into a different repeated sequence when generating its own history. This distinction helps explain why adding an EOS target is necessary but does not alone guarantee a well-formed reply. Keep generation budgets explicit in evaluation and saved artifacts.

Try it

Guided task 08

Preserve an instruction checkpoint and replay it

The trained tensors are only one part of an instruction model. Its character vocabulary, special-token IDs, formatting template and response-mask policy determine what those tensors mean. Save the architecture and update configuration alongside optimizer state, completed step count and random-generator state in a fresh temporary directory. This example records randomly initialized provenance so the artifact cannot be mistaken for a pretrained checkpoint.

Reload the plain-data checkpoint with weights_only enabled into a new compatible model. Compare the same validation logits in evaluation mode and regenerate the same bounded replies. The constructor recreates tied parameter identity before loading values. Exact replay here concerns this environment, fixed inputs and deterministic greedy inference. It does not claim bitwise resumed training across devices or library versions. The optimizer and RNG fields make later continuation experiments possible, while the verified inference comparison establishes that saving and loading retained the observed behavior. Keep final test outcomes out of configuration selection; the next lab defines a complete final evaluation and reporting contract.

Try it

  1. Save model state, optimizer state, progress and preparation metadata.
  2. Load a new compatible model from the fresh artifact.
  3. Compare logits and greedy continuations under the same evaluation inputs.

Extension 1

Optional real pretrained instruction fine-tuning

This standalone route loads the genuine GPT-2 checkpoint with its own subword tokenizer and unchanged architecture. Expect about 500 MB of pretrained weights, at least 8 GB of host RAM, and a longer CPU run than the tiny core; a GPU is preferable for larger experiments. Four short updates adapt only the final transformer block and final layer normalization. Earlier layers and the tied input/output embeddings remain frozen. The measured before/after values demonstrate mechanics on constructed held-out situations; they do not establish a useful general assistant. Prompt and space-prefixed response are encoded separately to preserve an explicit target boundary, and the complete response plus EOS must fit. The library shifts the full labels internally. Compare the same validation requests before and after adaptation under greedy decoding.

In the checked CPU run, response loss decreased while free responses still repeated parts of the prompt. This illustrates why teacher-forced loss and generated-response quality need separate evaluation.

Python · standalone extension
import json, tempfile
from pathlib import Path
import torch
from transformers import GPT2TokenizerFast,GPT2LMHeadModel
torch.set_num_threads(1)
torch.manual_seed(219)
checkpoint='openai-community/gpt2'
revision='607a30d783dfa663caf39e06633721c8d4cfcd7e'
tokenizer=GPT2TokenizerFast.from_pretrained(checkpoint,revision=revision)
tokenizer.pad_token=tokenizer.eos_token
tokenizer.padding_side='right'
model=GPT2LMHeadModel.from_pretrained(checkpoint,revision=revision,use_safetensors=True)
model.config.pad_token_id=tokenizer.eos_token_id
training=[('library','bring a book'),('park','bring water'),('rain','bring an umbrella'),('gym','bring a towel')]
validation=[('reading cafe','bring a book'),('workout','bring a towel')]
test=[('hot day','bring water'),('wet weather','bring an umbrella')]

def prompt(situation):
    return 'Task: name one useful item\nInput: '+situation+'\nReply:'

def batch(rows):
    segments=[]
    for situation,response in rows:
        if not response.strip():
            raise ValueError('Response must be nonempty.')
        prefix=tokenizer.encode(prompt(situation),add_special_tokens=False)
        answer=tokenizer.encode(' '+response.strip(),add_special_tokens=False)+[tokenizer.eos_token_id]
        if len(prefix)+len(answer)>96:
            raise ValueError('Complete reference and EOS must fit.')
        segments.append((prefix,answer))
    width=max(len(p)+len(a) for p,a in segments)
    ids=torch.full((len(rows),width),tokenizer.eos_token_id,dtype=torch.long)
    mask=torch.zeros_like(ids)
    labels=torch.full_like(ids,-100)
    for i,(prefix,answer) in enumerate(segments):
        full=prefix+answer
        ids[i,:len(full)]=torch.tensor(full)
        mask[i,:len(full)]=1
        labels[i,len(prefix):len(full)]=torch.tensor(answer)
    return {'input_ids':ids,'attention_mask':mask,'labels':labels}

def evaluate(rows):
    model.eval()
    total,count=0.0,0
    with torch.no_grad():
        for row in rows:
            data=batch([row])
            n=int((data['labels'][:,1:]!=-100).sum())
            # GPT2LMHeadModel shifts full labels internally: do not shift here.
            loss=model(**data,use_cache=False).loss
            total+=float(loss)*n
            count+=n
    return {'loss':total/count,'tokens':count,'records':len(rows)}

def replies(rows):
    model.eval()
    outputs=[]
    with torch.no_grad():
        for situation,reference in rows:
            data=tokenizer(prompt(situation),return_tensors='pt',add_special_tokens=False)
            # Generate one unpadded prompt at a time; retain its attention mask.
            generated=model.generate(**data,max_new_tokens=20,do_sample=False,pad_token_id=tokenizer.eos_token_id,eos_token_id=tokenizer.eos_token_id)
            new_ids=generated[0,data['input_ids'].shape[1]:]
            outputs.append({'input':situation,'reference':reference,'response':tokenizer.decode(new_ids,skip_special_tokens=True).strip(),'new_token_ids':new_ids.tolist()})
    return outputs

for parameter in model.parameters():
    parameter.requires_grad=False
for module in [model.transformer.h[-1],model.transformer.ln_f]:
    for parameter in module.parameters():
        parameter.requires_grad=True
trainable_names=[name for name,p in model.named_parameters() if p.requires_grad]
initial_overrides={name:p.detach().clone() for name,p in model.named_parameters() if p.requires_grad}
before=evaluate(validation)
before_replies=replies(validation)
optimizer=torch.optim.AdamW([p for p in model.parameters() if p.requires_grad],lr=0.00005)
training_batch=batch(training)
for step in range(4):
    model.train()
    optimizer.zero_grad(set_to_none=True)
    loss=model(**training_batch,use_cache=False).loss
    loss.backward()
    optimizer.step()
after=evaluate(validation)
after_replies=replies(validation)
print({'before':before,'after':after,'trainable_parameters':sum(p.numel() for p in model.parameters() if p.requires_grad),'steps':4})
print({'before_responses':before_replies,'after_responses':after_replies})
assert before['tokens']==after['tokens'] and torch.isfinite(loss)
assert any(not torch.equal(initial_overrides[name],parameter) for name,parameter in model.named_parameters() if parameter.requires_grad)

Extension 2

Optional: a low-rank adaptation layer

Appendix E introduces LoRA: keep a base weight matrix frozen and learn a factored additive update. In this standalone layer, input x has width 12, A has shape (12,2), and B has shape (2,8). The extra output is (x @ A @ B) × alpha/rank. The product has rank at most two; the base operation still runs and its original weights remain necessary. The adapter has 40 trainable scalars, compared with 96 in this bias-free base matrix. Savings depend on the chosen rank and layer dimensions.

Initialize B to zero so the adapter initially preserves the base output. During the first backward call, A has zero gradient through the zero B, while B can receive a nonzero gradient. Subsequent updates can train both factors. This exercise fits supplied numeric features to a simple numeric target; it demonstrates the adapter mechanism rather than language quality. A full GPT-2 adapter implementation must choose target projections, respect their weight layout and handle tied parameters correctly. This nn.Linear wrapper is not a drop-in replacement for every GPT-2 projection. Compare adaptation policies under the same held-out evaluation plan.

Python · standalone extension
import torch
from torch import nn
torch.manual_seed(719)
torch.set_num_threads(1)

class LowRankLinear(nn.Module):
    def __init__(self,input_width=12,output_width=8,rank=2,alpha=2.0):
        super().__init__()
        self.base=nn.Linear(input_width,output_width,bias=False)
        self.base.requires_grad_(False)
        self.a=nn.Parameter(torch.randn(input_width,rank)*0.05)
        self.b=nn.Parameter(torch.zeros(rank,output_width))
        self.scale=alpha/rank
    def forward(self,x):
        return self.base(x)+(x@self.a@self.b)*self.scale

layer=LowRankLinear()
x=torch.randn(6,12)
target=0.2*x[:,:8]  # A numeric mechanics target, not a language task.
base_before=layer.base.weight.detach().clone()
with torch.no_grad():
    assert torch.equal(layer(x),layer.base(x))
optimizer=torch.optim.AdamW([layer.a,layer.b],lr=0.02,weight_decay=0)
losses=[]
for step in range(20):
    optimizer.zero_grad(set_to_none=True)
    loss=(layer(x)-target).square().mean()
    loss.backward()
    if step==0:
        assert torch.count_nonzero(layer.a.grad)==0
        assert layer.b.grad.norm()>0
    optimizer.step()
    losses.append(float(loss.detach()))
with torch.no_grad():
    final_loss=float((layer(x)-target).square().mean())
trainable=sum(p.numel() for p in layer.parameters() if p.requires_grad)
assert trainable==40 and torch.equal(base_before,layer.base.weight)
assert layer.base.weight.grad is None and torch.count_nonzero(layer.b)>0
print({'adapter_parameters':trainable,'frozen_base_parameters':base_before.numel(),
       'initial_loss':losses[0],'final_loss':final_loss,'steps':20})
