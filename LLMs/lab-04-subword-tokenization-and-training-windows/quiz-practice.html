# A minimal byte-pair learner: no downloads and no claim of GPT-2 compatibility.
def merge_pair(sequence,pair,new_id):
    output = []
    position = 0
    while position<len(sequence):
        if position+1<len(sequence) and (sequence[position],sequence[position+1])==pair:
            output.append(new_id)
            position += 2
        else:
            output.append(sequence[position])
            position += 1
    return output

def fit_byte_merges(texts,num_merges):
    vocabulary = {index:bytes([index]) for index in range(256)}
    sequences = [list(text.encode("utf-8")) for text in texts]
    merges,history = [],[]
    for step in range(num_merges):
        counts = Counter(pair for sequence in sequences for pair in zip(sequence[:-1],sequence[1:]))
        if not counts:
            break
        pair = min(counts,key=lambda candidate:(-counts[candidate],candidate))
        if counts[pair]<2:
            break
        new_id = len(vocabulary)
        vocabulary[new_id] = vocabulary[pair[0]]+vocabulary[pair[1]]
        before = sum(map(len,sequences))
        sequences = [merge_pair(sequence,pair,new_id) for sequence in sequences]
        after = sum(map(len,sequences))
        merges.append((pair,new_id))
        history.append({"pair":pair,"new_id":new_id,"frequency":counts[pair],"before":before,"after":after})
    return vocabulary,merges,history

def encode_bytes(text,merges):
    sequence = list(text.encode("utf-8"))
    for pair,new_id in merges:
        sequence = merge_pair(sequence,pair,new_id)
    return sequence

def decode_bytes(ids,vocabulary):
    return b"".join(vocabulary[index] for index in ids).decode("utf-8")
