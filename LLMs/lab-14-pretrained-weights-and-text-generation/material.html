import math

import torch
from torch import nn
from torch.nn import functional as F
import matplotlib.pyplot as plt
torch.set_num_threads(1)
torch.manual_seed(114)
rng = np.random.default_rng(114)
def as_array(tensor):
    return tensor.detach().cpu().numpy()
class CausalHeads(nn.Module):
    def __init__(self, width=16, heads=4, dropout=0.0):
        super().__init__()
        if width % heads:
            raise ValueError("width must be divisible by heads")
        self.width, self.heads, self.head_dim = width, heads, width // heads
        self.qkv = nn.Linear(width, 3*width, bias=False)
        self.proj = nn.Linear(width, width, bias=False)
        self.attn_drop = nn.Dropout(dropout)
    def forward(self, x):
        b,t,c = x.shape
        packed = self.qkv(x).reshape(b,t,3,self.heads,self.head_dim)
        q,k,v = packed.permute(2,0,3,1,4).unbind(0)
        scores = (q @ k.transpose(-2,-1)) / math.sqrt(self.head_dim)
        future = torch.triu(torch.ones(t,t,device=x.device,dtype=torch.bool),diagonal=1)
        weights = scores.masked_fill(future,float("-inf")).softmax(dim=-1)
        context = self.attn_drop(weights) @ v
        merged = context.transpose(1,2).contiguous().reshape(b,t,c)
        return self.proj(merged), weights
class FeedForward(nn.Module):
    def __init__(self,width=16,expansion=4):
        super().__init__()
        self.up = nn.Linear(width,expansion*width)
        self.act = nn.GELU()
        self.down = nn.Linear(expansion*width,width)
    def forward(self,x):
        return self.down(self.act(self.up(x)))
class TransformerBlock(nn.Module):
    def __init__(self,width=16,heads=4,dropout=0.0):
        super().__init__()
        self.norm1 = nn.LayerNorm(width)
        self.attention = CausalHeads(width,heads,dropout)
        self.norm2 = nn.LayerNorm(width)
        self.ff = FeedForward(width)
        self.drop = nn.Dropout(dropout)
    def forward(self,x):
        x = x + self.drop(self.attention(self.norm1(x))[0])
        return x + self.drop(self.ff(self.norm2(x)))
class TinyGPT(nn.Module):
    def __init__(self,vocab_size,context=12,width=16,heads=4,layers=2,dropout=0.0):
        super().__init__()
        self.config = dict(vocab_size=vocab_size,context=context,width=width,heads=heads,layers=layers,dropout=dropout)
        self.token = nn.Embedding(vocab_size,width)
        self.position = nn.Embedding(context,width)
        self.drop = nn.Dropout(dropout)
        self.blocks = nn.ModuleList([TransformerBlock(width,heads,dropout) for _ in range(layers)])
        self.final_norm = nn.LayerNorm(width)
    def forward(self,ids):
        if ids.ndim != 2 or ids.shape[1] == 0 or ids.shape[1] > self.config["context"]:
            raise ValueError("Expected nonempty B,T IDs within context")
        positions = torch.arange(ids.shape[1],device=ids.device)
        x = self.drop(self.token(ids) + self.position(positions)[None])
        for block in self.blocks:
            x = block(x)
        x = self.final_norm(x)
        return F.linear(x,self.token.weight)  # Tied input/output weights.
vocab = ["<pad>","<eos>","<unk>","library","opens","at","nine","ten","room","is","free","busy","please","return","the","book","today","tomorrow","desk","closes","six","seven","reserve","a","quiet","seat","now"]
stoi = {word:i for i,word in enumerate(vocab)}
def encode(text,eos=False):
    ids = [stoi.get(word,stoi["<unk>"]) for word in text.lower().split()]
    return ids + ([stoi["<eos>"]] if eos else [])
def decode(ids):
    return " ".join(vocab[int(i)] for i in ids)
train_documents = ["library opens at nine","please return the book today","room is free today","reserve a quiet seat","desk closes at six","room is busy now"]
validation_documents = ["library closes at seven","please return the book tomorrow"]
test_documents = ["room is free tomorrow","desk opens at ten"]
# Shift exactly once. Padding is outside each document's true target positions.
def make_batch(documents):
    sequences = [encode(text,eos=True) for text in documents]
    if not sequences or any(len(sequence)<2 for sequence in sequences):
        raise ValueError("Need nonempty documents with at least one word")
    lengths = torch.tensor([len(sequence) for sequence in sequences])
    padded = torch.full((len(sequences),int(lengths.max())),stoi["<pad>"],dtype=torch.long)
    for row,sequence in enumerate(sequences):
        padded[row,:len(sequence)] = torch.tensor(sequence)
    inputs = padded[:,:-1].clone()
    targets = padded[:,1:].clone()
    valid = torch.arange(targets.shape[1])[None] < (lengths-1)[:,None]
    targets[~valid] = -100
    return inputs,targets

def token_loss(model,inputs,targets):
    logits = model(inputs)
    return F.cross_entropy(logits.reshape(-1,len(vocab)),targets.reshape(-1),ignore_index=-100)
def update(model,optimizer,inputs,targets):
    model.train()
    optimizer.zero_grad(set_to_none=True)
    loss = token_loss(model,inputs,targets)
    loss.backward()
    optimizer.step()
    return float(loss.detach())
def measure(model,documents):
    previous = model.training
    model.eval()
    loss_sum,count = 0.0,0
    try:
        with torch.no_grad():
            for text in documents:
                inputs,targets = make_batch([text])
                loss_sum += float(F.cross_entropy(model(inputs).reshape(-1,len(vocab)),targets.reshape(-1),ignore_index=-100,reduction="sum"))
                count += int((targets!=-100).sum())
    finally:
        model.train(previous)
    return loss_sum/count

Guided task 01

Checkpoint and architecture compatibility

A model becomes pretrained because its weights were learned previously, not because its class has GPT in the name. The offline model in this lesson starts randomly and learns only a few constructed notices. The later extension loads an actual GPT-2 checkpoint with its original tokenizer and architecture. These are different artifacts with different capabilities and resource requirements.

Compatibility includes vocabulary ordering, layer definitions, dimensions and tensor layout. Even two models with the same parameter count need not represent the same computation. Inspect two tiny configurations to see how one architectural change affects stored shapes. Renaming checkpoint keys or suppressing missing-key errors cannot turn a width-8 teaching model into the architecture expected by the official GPT-2 weights. Use the matching public API for that checkpoint instead.

  2. Inspect their token-embedding and packed attention-projection shapes.
  3. Plot their actual parameter counts, counting tied embeddings once.

Guided task 02

A learned local distribution

Train the smaller model briefly so later decoding experiments use learned parameters rather than unexplained random scores. The training documents are the same small notice fixtures used earlier. The task compares the next-token distribution for the prefix 'library opens at' before and after forty updates. Because that prefix occurs in training, this is an illustration of fitting, not held-out evaluation.

The final prefix position supplies the distribution for the next token. Earlier positions predict continuations of shorter prefixes and should not be averaged together for this decision. Softmax is useful for displaying probabilities; the training loss still receives raw logits. The offline model remains limited to its fixed word vocabulary, including an unknown-token entry for unseen words.

Try it

Guided task 03

Bounded greedy decoding

Greedy decoding chooses the largest current logit, appends that token and recomputes the next decision. It needs no softmax because softmax preserves logit order. The procedure is deterministic for this evaluation-mode CPU model, but it does not search for the globally most likely complete sentence. A locally preferred token changes the context used for every later decision.

Our helper accepts one nonempty prompt and a nonnegative new-token budget. It keeps the full returned sequence but sends only the most recent context window into the model. TinyGPT resets its learned position indices within that window; this is an explicit educational policy, not an unlimited-memory claim. Stop after appending EOS or when the budget is exhausted. A prompt already ending in EOS is treated as complete.

Try it

  1. Define a greedy helper with the stated shape and stopping contract.
  2. Generate at most six new tokens after 'please return'.
  3. Print the continuation and plot the cumulative returned length.
Python · run in order
def greedy(model,prompt,max_new):
    if prompt.ndim!=2 or prompt.shape[0]!=1 or prompt.shape[1]==0 or type(max_new) is not int or max_new<0:
        raise ValueError("Need one nonempty prompt and a nonnegative integer budget")
    result = prompt.clone()
    previous = model.training
    model.eval()
    try:
        with torch.no_grad():
            for unused in range(max_new):
                if int(result[0,-1])==stoi["<eos>"]:
                    break
                visible = result[:,-model.config["context"]:]
                chosen = model(visible)[:,-1].argmax(dim=-1,keepdim=True)
                result = torch.cat([result,chosen],dim=1)
    finally:
        model.train(previous)
    return result
prompt = torch.tensor([encode("please return")])
greedy_ids = greedy(model,prompt,6)
print(decode(greedy_ids[0]))
lengths = list(range(prompt.shape[1],greedy_ids.shape[1]+1))
fig,ax = plt.subplots(figsize=(6,3.4))
ax.plot(range(len(lengths)),lengths,"o-")
ax.set(xlabel="New tokens appended",ylabel="Returned sequence length",title="Bounded greedy continuation")
ax.set_xticks(range(len(lengths)))
ax.set_yticks(lengths)
fig.tight_layout()

Read the result

The output always begins with the original prompt. EOS, when generated, remains in the returned IDs so the stopping event can be inspected. The plotted length can stop before the six-token budget, and the procedure never depends on an unbounded while loop.

  • Supply a prompt longer than the context window and inspect which suffix the model actually sees.

Guided task 04

Temperature on logits

Temperature divides logits before softmax. For a positive temperature below one, differences between logits become more pronounced; above one, the distribution becomes flatter. The ordering of unequal logits does not change. Consequently, changing positive temperature alone does not change a greedy argmax choice, although it changes probabilities used by sampling.

Use the same learned next-token logits at three temperatures so the comparison isolates decoding rather than training. A temperature of zero is not a valid divisor; request greedy decoding explicitly instead. Extremely small positive values can produce an almost one-hot distribution, while extremely large values approach uniformity over the finite vocabulary. Neither behavior establishes factual accuracy or adds knowledge to the model.

Try it

  1. Compute probabilities at temperatures 0.6, 1.0 and 1.8.
  2. Measure each distribution's entropy in natural-log units.
  3. Plot the same six token IDs at every temperature.

Guided task 05

Top-k support and renormalization

Top-k filtering limits sampling to a fixed number of leading logits. It is a support restriction: excluded entries receive zero probability after their logits become negative infinity. The retained entries are then renormalized together. Selecting the top probabilities after sampling would be a different operation and would not implement this policy.

The helper below retains exactly k indices returned by topk. A threshold-based implementation can retain more than k entries when several logits tie at the cutoff. Here a tie is resolved by the library's chosen indices; do not assume a portable ordering among tied values. Positive temperature preserves rankings, so the support chosen from the original logits also serves the temperature-scaled distribution. We use one-dimensional next-token logits to keep every axis explicit.

Try it

  1. Validate k as an integer between 1 and the vocabulary size.
  2. Keep exactly four leading logits and normalize at temperature 0.9.
  3. Compare the full vocabulary support before and after filtering.
Python · run in order
def topk_probabilities(logits,k,temperature=1.0):
    if logits.ndim!=1 or type(k) is not int or not 1<=k<=logits.numel():
        raise ValueError("Invalid logits shape or k")
    if not math.isfinite(temperature) or temperature<=0:
        raise ValueError("Temperature must be positive and finite")
    indices = torch.topk(logits,k).indices
    filtered = torch.full_like(logits,float("-inf"))
    filtered[indices] = logits[indices]/temperature
    return filtered.softmax(dim=-1)
filtered_probabilities = topk_probabilities(next_logits,4,0.9)
fig,ax = plt.subplots(figsize=(7,3.7))
ax.plot(range(len(vocab)),as_array((next_logits/0.9).softmax(dim=-1)),"o-",label="All tokens")
ax.stem(range(len(vocab)),as_array(filtered_probabilities),linefmt="C1-",markerfmt="C1s",basefmt=" ")
ax.plot([],[],"s",color="C1",label="Top 4")
ax.set(xlabel="Vocabulary ID",ylabel="Probability",title="Filtering changes support and normalization")
ax.set_xticks(range(0,len(vocab),3))
ax.legend()
fig.tight_layout()

Read the result

Only retained tokens have positive probability. Their values usually increase relative to the unfiltered distribution because excluded probability mass is removed before normalization. When k equals one, the distribution becomes concentrated on one selected maximum; when k equals the vocabulary size, filtering removes nothing.

Change and compare

  • Set k to the full vocabulary size and compare with ordinary temperature-scaled softmax.
  • Use tied controlled logits in a separate trial and count retained indices without asserting a particular tie order.

Guided task 06

Sampling with a dedicated generator

Sampling draws a token according to probabilities instead of always selecting the maximum. A seed controls a random stream, not the meaning or quality of the generated text. Use a dedicated CPU torch.Generator so unrelated random draws elsewhere in the notebook do not change this sampling stream. The same model, prompt, settings and generator state reproduce the same decisions in this controlled environment.

The sampling loop shares greedy decoding's context and stopping policies. At each step, recompute logits from the current suffix, filter and normalize them, then use multinomial for one draw. The helper restores the caller's training mode when finished while disabling dropout and autograd during generation. Repeat the full call with newly seeded generators rather than reusing one already advanced by a previous call.

Try it

  1. Implement single-prompt sampling with a finite budget and top-k filtering.
  2. Run the same prefix twice with separate generators seeded 1414.
  3. Print both results and plot the IDs actually drawn.
Python · run in order
def sampled(model,prompt,max_new,k,temperature,generator):
    if prompt.ndim!=2 or prompt.shape[0]!=1 or prompt.shape[1]==0 or type(max_new) is not int or max_new<0:
        raise ValueError("Invalid prompt or budget")
    result = prompt.clone()
    previous = model.training
    model.eval()
    try:
        with torch.no_grad():
            for unused in range(max_new):
                if int(result[0,-1])==stoi["<eos>"]:
                    break
                logits = model(result[:,-model.config["context"]:])[0,-1]
                probabilities = topk_probabilities(logits,k,temperature)
                chosen = torch.multinomial(probabilities,1,generator=generator).reshape(1,1)
                result = torch.cat([result,chosen],dim=1)
    finally:
        model.train(previous)
    return result
sample_prompt = torch.tensor([encode("room is")])
draw_a = sampled(model,sample_prompt,7,4,1.2,torch.Generator().manual_seed(1414))
draw_b = sampled(model,sample_prompt,7,4,1.2,torch.Generator().manual_seed(1414))
print(decode(draw_a[0]))
print(decode(draw_b[0]))
drawn = draw_a[0,sample_prompt.shape[1]:]
fig,ax = plt.subplots(figsize=(6.6,3.4))
ax.plot(range(1,len(drawn)+1),as_array(drawn),"o-")
ax.set(xlabel="Generated position",ylabel="Selected token ID",title="A reproducible sampled trajectory")
ax.set_xticks(range(1,len(drawn)+1))
ax.set_yticks(sorted(set(drawn.tolist())))
fig.tight_layout()

Read the result

The repeated outputs should match exactly here. Different seeds may still yield the same continuation when the distribution is concentrated, so diversity is not guaranteed by changing a seed. Record the checkpoint identity and decoding settings alongside it; a seed alone does not reproduce an experiment with different weights.

Guided task 07

Stopping rules as a testable

A generation loop needs explicit behavior at boundaries, even when a learned model rarely produces them in a short demonstration. Test the existing greedy helper using controlled scripted logits. This small test object is not a trained language model: it deliberately makes a known token the unique maximum at each call. That lets us check stopping logic independently from uncertain learned predictions.

The contract includes EOS in the returned record, appends nothing after it and returns an already complete prompt unchanged. A zero-token budget must perform no model calls. The budget also bounds a model that never predicts EOS. These are behavior tests, not evidence that generated messages are meaningful. Keeping token-policy mechanics separate from quality judgments makes failures easier to diagnose.

Try it

  1. Create a scripted predictor whose next choices are seat, EOS and now.
  2. Run the greedy helper with a five-token budget.
  3. Compare appended-token counts for EOS, zero-budget and already-complete cases.
Python · run in order
class ScriptedPredictor(nn.Module):
    def __init__(self,choices):
        super().__init__()
        self.choices = choices
        self.calls = 0
        self.config = {"context":4}
    def forward(self,ids):
        selected_id = self.choices[min(self.calls,len(self.choices)-1)]
        self.calls += 1
        logits = torch.full((1,ids.shape[1],len(vocab)),-10.0)
        logits[0,-1,selected_id] = 10.0
        return logits
test_prompt = torch.tensor([encode("reserve a")])
scripted = ScriptedPredictor([stoi["seat"],stoi["<eos>"],stoi["now"]])
stopped = greedy(scripted,test_prompt,5)
no_budget_model = ScriptedPredictor([stoi["seat"]])
unchanged = greedy(no_budget_model,test_prompt,0)
complete_model = ScriptedPredictor([stoi["seat"]])
complete_prompt = torch.tensor([[stoi["<eos>"]]])
already_complete = greedy(complete_model,complete_prompt,5)
appended_counts = [stopped.shape[1]-2,unchanged.shape[1]-2,already_complete.shape[1]-1]
fig,ax = plt.subplots(figsize=(7,3.4))
ax.bar(['EOS reached','Zero budget','Prompt ends EOS'],appended_counts)
ax.set(title="Controlled stopping tests",ylabel="Appended tokens")
fig.tight_layout()
ax.set_yticks(range(max(appended_counts)+1))

Read the result

The scripted predictor makes EOS behavior reproducible: no later scripted token should appear. Its call counter verifies that the zero-budget and already-complete cases avoid unnecessary forward passes. Test fixtures like this complement real-model examples because the critical edge case is guaranteed to occur.

Change and compare

  • Replace the scripted choices with one non-EOS token and confirm the budget stops generation.
  • Test a one-token budget when EOS would otherwise be produced on the second call.

Guided task 08

A reproducible generation artifact

A reproducible continuation depends on more than the model weights. Save the configuration, vocabulary, tokenization convention, prompt and decoding policy together. For sampled output, also retain the generator seed or state; this task uses greedy decoding, so no sampling state is needed. The saved record describes the small locally trained model, not the real pretrained checkpoint used in the separate extension.

Reload into a fresh matching model and compare the generated IDs. This checks the whole path from token IDs through the policy, rather than only one forward pass. It does not establish factual correctness, politeness or suitability for a student-facing service. A later evaluation should separately ask whether the text follows the intended task and whether any factual claim is supported by an authoritative source.

Extension 1

Standalone extension: real pretrained GPT-2

This separate route loads the real openai-community/gpt2 checkpoint through its matching Transformers classes. It does not depend on the offline setup and does not place GPT-2 tensors into TinyGPT. Install torch and transformers 4.57.6 in your chosen environment first. The initial checkpoint download is approximately 500 MB; plan for at least 8 GB host RAM, with a GPU useful for longer experiments. No paid API is needed. The pinned revision identifies the exact model/tokenizer files.

The first forward pass scores two short constructed notices. Use right padding for this loss example, append a genuine EOS, copy input_ids into labels, and ignore only positions where attention_mask is zero. GPT2LMHeadModel shifts internally, so the code does not pre-shift its labels. The manual loss comparison below shifts only for verification of the returned logits. It also counts the actual supervised shifted tokens.

For batched decoder-only generation, switch the tokenizer to left padding so the final column is a real prompt token in each row; generate uses the attention mask to manage visible positions. Keep the checkpoint architecture intact, use evaluation mode and no_grad, and request at most 12 new tokens. The output is an English continuation from pretrained weights, not a verified campus notice or an instruction-tuned assistant response.

Python · standalone extension
import math
import torch
from transformers import GPT2TokenizerFast,GPT2LMHeadModel
torch.set_num_threads(1)
torch.manual_seed(114)
model_id = "openai-community/gpt2"
revision = "607a30d783dfa663caf39e06633721c8d4cfcd7e"
tokenizer = GPT2TokenizerFast.from_pretrained(model_id,revision=revision)
tokenizer.pad_token = tokenizer.eos_token
model = GPT2LMHeadModel.from_pretrained(model_id,revision=revision,use_safetensors=True)
model.config.pad_token_id = tokenizer.pad_token_id
model.eval()
print("Pretrained parameters:",sum(p.numel() for p in model.parameters()))
print("Architecture:",model.config.n_layer,model.config.n_head,model.config.n_embd)
# Score fixed notices. Genuine EOS positions remain supervised.
tokenizer.padding_side = "right"
notices = ["The library opens at nine."+tokenizer.eos_token,"Please return your borrowed book tomorrow."+tokenizer.eos_token]
batch = tokenizer(notices,padding=True,truncation=True,max_length=48,add_special_tokens=False,return_tensors="pt")
labels = batch["input_ids"].clone()
labels[batch["attention_mask"]==0] = -100
with torch.no_grad():
    scored = model(**batch,labels=labels)
shifted_labels = labels[:,1:]
supervised_tokens = int((shifted_labels!=-100).sum())
manual_loss = torch.nn.functional.cross_entropy(scored.logits[:,:-1].contiguous().view(-1,model.config.vocab_size),shifted_labels.contiguous().view(-1),ignore_index=-100)
assert supervised_tokens>0
assert torch.allclose(scored.loss,manual_loss,atol=1e-5)
assert int(((labels==tokenizer.eos_token_id)&batch["attention_mask"].bool()).sum())==2
print("Mean supervised token loss:",float(scored.loss),"Token count:",supervised_tokens)
# Left padding makes the final input column a real token for generation.
tokenizer.padding_side = "left"
prompts = ["The library opens at","Please return"]
request = tokenizer(prompts,padding=True,truncation=True,max_length=32,return_tensors="pt")
with torch.no_grad():
    generated = model.generate(**request,max_new_tokens=12,do_sample=False,pad_token_id=tokenizer.pad_token_id,eos_token_id=tokenizer.eos_token_id)
continuations = generated[:,request["input_ids"].shape[1]:]
for prompt,ids in zip(prompts,continuations):
    print("Prompt:",prompt)
    print("Continuation:",tokenizer.decode(ids,skip_special_tokens=True))
assert generated.shape[1]<=request["input_ids"].shape[1]+12
assert all(int(row[-1])==1 for row in request["attention_mask"])
assert math.isfinite(float(scored.loss))
