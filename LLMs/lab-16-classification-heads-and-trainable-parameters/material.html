Classification Heads and Trainable Parameters

MaterialPracticeQuiz

Connect a causal text representation to a small action classifier. Inspect what the head sees, which parameters can change, and how padding affects the result.

What you will learn

  • Pool the final real token from right-padded sequences.
  • Map contextual features to class logits and probabilities.
  • Verify frozen and trainable parameter groups.
  • Distinguish evaluation mode from disabled gradient recording.

Before this lab: Use Python, token IDs, tensor axes and the causal-transformer concepts from preceding labs.

In this lab

  1. Contextual representations precede the head
  2. Pool from the last valid token
  3. A linear head produces class logits
  4. Freeze parameters explicitly
  5. Cross entropy uses one class target per row
  6. Verify the update boundary with gradients
  7. Test invariance to extra right padding
  8. Evaluation mode is not gradient disabling

Python · shared setup
import copy, json, re, tempfile
from pathlib import Path
import numpy as np
import torch
from torch import nn
from torch.nn import functional as F
import matplotlib.pyplot as plt
torch.set_num_threads(1)
torch.manual_seed(116)
# Each pair describes the same room request; group before splitting.
pairs = [
    (0, ('reserve a quiet room', 'please reserve the quiet room')),
    (1, ('cancel my quiet room', 'please cancel the quiet room')),
    (0, ('reserve a large room', 'please reserve the large room')),
    (1, ('cancel my large room', 'please cancel the large room')),
    (0, ('reserve a room today', 'please reserve my room today')),
    (1, ('cancel a room today', 'please cancel my room today')),
    (0, ('reserve the room tomorrow', 'please reserve a room tomorrow')),
    (1, ('cancel the room tomorrow', 'please cancel a room tomorrow')),
    (0, ('reserve a sunny room', 'please reserve the sunny room')),
    (1, ('cancel my sunny room', 'please cancel the sunny room')),
    (0, ('reserve a corner room', 'please reserve the corner room')),
    (1, ('cancel my corner room', 'please cancel the corner room')),
]
records = [dict(id=f'g{g}_v{v}', group=g, text=text, label=label)
           for g, (label, texts) in enumerate(pairs) for v, text in enumerate(texts)]
split_groups = {'train': set(range(8)), 'validation': {8, 9}, 'test': {10, 11}}
def words(text):
    return re.findall(r"[a-z]+", text.lower())
# Vocabulary is fitted on training records only; IDs 0/1 mean PAD/UNK.
training_words = sorted({word for r in records if r['group'] in split_groups['train'] for word in words(r['text'])})
vocab = {'<pad>': 0, '<unk>': 1, **{word: i+2 for i, word in enumerate(training_words)}}
def encode(text, limit=12):
    ids = [vocab.get(word, 1) for word in words(text)][:limit]
    if not ids:
        raise ValueError('A message must contain at least one token.')
    return ids

def make_batch(rows, limit=12):
    if not rows:
        raise ValueError('Cannot collate an empty batch.')
    sequences = [encode(r['text'], limit) for r in rows]
    lengths = torch.tensor([len(s) for s in sequences], dtype=torch.long)
    ids = torch.zeros((len(rows), int(lengths.max())), dtype=torch.long)
    for i, sequence in enumerate(sequences):
        ids[i, :len(sequence)] = torch.tensor(sequence)
    valid = torch.arange(ids.shape[1])[None, :] < lengths[:, None]
    labels = torch.tensor([r['label'] for r in rows], dtype=torch.long)
    return ids, valid, labels
# One small causal transformer block, randomly initialized: not pretrained.
class TinyBackbone(nn.Module):
    def __init__(self, vocabulary_size, width=24, context=32):
        super().__init__()
        self.token = nn.Embedding(vocabulary_size, width)
        self.position = nn.Embedding(context, width)
        self.norm1 = nn.LayerNorm(width)
        self.attention = nn.MultiheadAttention(width, 4, dropout=0.1, batch_first=True)
        self.norm2 = nn.LayerNorm(width)
        self.ff = nn.Sequential(nn.Linear(width, width*2), nn.GELU(), nn.Linear(width*2, width))
        self.final_norm = nn.LayerNorm(width)
    def forward(self, ids, valid):
        if ids.ndim != 2 or ids.shape != valid.shape or ids.shape[1] > self.position.num_embeddings:
            raise ValueError('Invalid token shape or context length.')
        if not torch.all(valid.any(dim=1)):
            raise ValueError('Every row needs a real token.')
        length = ids.shape[1]
        x = self.token(ids) + self.position(torch.arange(length, device=ids.device))[None]
        causal = torch.triu(torch.ones(length, length, dtype=torch.bool, device=ids.device), diagonal=1)
        normalized = self.norm1(x)
        attended, _ = self.attention(normalized, normalized, normalized, attn_mask=causal, key_padding_mask=~valid, need_weights=False)
        x = x + attended
        return self.final_norm(x + self.ff(self.norm2(x)))

class TinyClassifier(nn.Module):
    def __init__(self):
        super().__init__()
        self.backbone = TinyBackbone(len(vocab))
        self.head = nn.Linear(24, 2)
    def forward(self, ids, valid):
        hidden = self.backbone(ids, valid)
        last = valid.sum(dim=1) - 1  # Right padding, never select a padded position.
        pooled = hidden[torch.arange(ids.shape[0], device=ids.device), last]
        return self.head(pooled)
# Small plotting helpers display measured tensors and counts.
def bar_plot(labels, values, title, ylabel):
    fig, ax = plt.subplots(figsize=(7, 3.5))
    ax.bar(labels, np.asarray(values))
    ax.set(title=title, ylabel=ylabel)
    if np.all(np.asarray(values)==0):
        ax.set_ylim(0,1)
        ax.text(0.5,0.5,'All values are zero',transform=ax.transAxes,ha='center')
    fig.tight_layout()
    return fig

def heat_plot(values, title, xlabel, ylabel, label):
    array = values.detach().cpu().numpy() if isinstance(values, torch.Tensor) else np.asarray(values)
    fig, ax = plt.subplots(figsize=(7, 3.5))
    if np.any(array < 0):
        bound = max(float(np.abs(array).max()), 1e-8)
        view = ax.imshow(array, aspect='auto', interpolation='nearest', cmap='coolwarm', vmin=-bound, vmax=bound)
    else:
        view = ax.imshow(array, aspect='auto', interpolation='nearest', cmap='viridis', vmin=0, vmax=max(float(array.max()),1e-8))
    ax.set_xticks(np.unique(np.linspace(0,array.shape[1]-1,min(8,array.shape[1])).astype(int)))
    ax.set_yticks(np.unique(np.linspace(0,array.shape[0]-1,min(8,array.shape[0])).astype(int)))
    ax.set(title=title, xlabel=xlabel, ylabel=ylabel)
    fig.colorbar(view, ax=ax, label=label)
    fig.tight_layout()
    return fig

Guided task 01

Contextual representations precede the head

The shared setup contains a small causal transformer, initialized randomly for this lesson. Token and position embeddings enter one pre-normalized attention block and a feed-forward block, followed by a final normalization. The output has one feature vector per stored token position. It is not a pretrained language representation, and its values should not be described as learned understanding of room requests.

Classification adds a separate decision stage to these representations. First inspect the backbone output without considering labels. The tensor axes are batch, position and feature width. Causal attention prevents a real token from using later content, and the key-padding mask excludes padding positions as sources. Padded query positions may still have computed vectors, but they are not valid summary positions. Layer normalization acts on each feature vector's last dimension; it does not replace the mask. The visible feature values provide a useful shape and arithmetic check, not semantic interpretations of individual feature coordinates. A genuine pretrained route appears after the core tasks with its own checkpoint-compatible architecture.

Try it

  1. Create a fresh tiny classifier in evaluation mode.
  2. Pass two requests through its backbone with a boolean valid mask.
  3. Plot feature coordinates at real positions of the first message.
Python · run in order
rows = [records[0],records[3]]
ids,valid,labels = make_batch(rows)
model = TinyClassifier().eval()
with torch.no_grad():
    hidden = model.backbone(ids,valid)
fig = heat_plot(hidden[0,:int(valid[0].sum()),:8], 'Measured contextual feature coordinates', 'Feature coordinate 0–7', 'Real token position', 'Activation')

Read the result

The feature width is fixed while sequence length varies. Treat the values as randomly initialized model activations, not a map of human concepts.

  • Check the full hidden tensor shape.
  • Explain why padding queries may have values even though padding is not message content.

Guided task 02

Pool from the last valid token

A causal representation at the final real token can incorporate the retained message prefix. Right padding makes that position different for each row. Compute the last index from the attention mask, then gather one vector per message. Selecting hidden[:, -1] instead would use the batch's final storage column, which may contain padding for shorter messages.

The gather operation pairs a row index with that row's last valid position. Both index arrays have batch length, so the result has shape batch by feature width rather than creating all combinations of rows and positions. This is an important distinction when translating indexing between NumPy and PyTorch. The formula assumes nonempty right-padded sequences. Left padding, holes in a mask, or zero-length messages require a different contract and must not silently enter this helper. Pooling the last real token is a declared design choice, not proof that it is optimal for every classification task. Its immediate advantage here is that adding unused storage at the right cannot intentionally redefine which message representation reaches the head.

Try it

  1. Compute last indices as real lengths minus one.
  2. Gather one hidden vector for each batch row.
  3. Compare with blind final-column pooling on the shorter message.
Python · run in order
last = valid.sum(1)-1
pooled = hidden[torch.arange(len(rows)),last]
blind = hidden[:,-1]
pool_difference = (pooled-blind).norm(dim=1)
fig = bar_plot([r['id'] for r in rows],pool_difference.numpy(),'Last-valid versus final-storage pooling','Feature-vector distance')

Read the result

A zero distance for the longest row does not validate blind pooling for shorter rows. The correct coordinate comes from the mask for that row, not from batch width.

Change and compare

  • Add three padding columns and predict the correct last indices.
  • Trace paired indexing using a tiny hand-labeled tensor.

Guided task 03

A linear head produces class logits

The classification head maps a message vector to two unnormalized scores. Its weight matrix has one row per class and one column per feature, plus one bias per class. The class order remains reserve then cancel, matching the dataset. Replacing a language-model vocabulary projection with this head changes the output's meaning: these two coordinates describe actions, not next-token IDs.

Softmax converts the logits into a normalized distribution for inspection, while argmax selects the larger score. Cross-entropy training should receive the raw logits rather than probabilities that have already passed through softmax. A sum of one is a mathematical property of the transformation and does not establish calibrated confidence. This head is newly initialized; a high value in its probability plot should not be treated as evidence of competence. Inspect how the batch dimension stays aligned with the label vector and how the feature dimension disappears into the linear map. A later adaptation run will update selected parameters using examples, then compare held-out behavior separately.

Try it

  1. Apply the two-output linear head to pooled features.
  2. Compute probabilities along the class axis.
  3. Inspect the per-message probability matrix and selected labels.
Python · run in order
logits = model.head(pooled)
probabilities = torch.softmax(logits,dim=1)
predicted = logits.argmax(1)
fig = heat_plot(probabilities,'New head: uncalibrated probabilities','Class 0 reserve, 1 cancel','Message row','Softmax value')

Read the result

Every row sums to one even though the head has not learned the task. Normalization and correctness are different properties.

Change and compare

  • Compute the head parameter count from its dimensions.
  • Add the same scalar to both logits and compare the probabilities.

Guided task 04

Freeze parameters explicitly

A parameter's requires_grad flag controls whether autograd accumulates its gradient. Freezing the backbone and training only the head reduces the number of updated parameters, but does not make the randomly initialized backbone pretrained. In a real transfer-learning route, those frozen representations come from a previously trained checkpoint. The two situations may share code while supporting very different performance expectations.

Set the flags first, then create an optimizer from the trainable parameters. Listing names and counts makes the policy reviewable. A frozen module still performs its forward calculation, and its dropout behavior is controlled separately by train or eval mode. If fixed features are intended, keep backbone dropout disabled rather than assuming requires_grad does that automatically. Freezing also does not remove the storage occupied by the base model. A small count of trainable parameters should therefore not be advertised as the total memory footprint. This task records both the full parameter count and the updated subset before any optimization step.

Try it

  1. Freeze every backbone parameter and leave the head trainable.
  2. Count total and trainable scalar parameters.
  3. Inspect the trainable parameter names before creating the optimizer.
Python · run in order
for parameter in model.parameters():
    parameter.requires_grad = False
for parameter in model.head.parameters():
    parameter.requires_grad = True
trainable_names = [name for name,p in model.named_parameters() if p.requires_grad]
total_count = sum(p.numel() for p in model.parameters())
trainable_count = sum(p.numel() for p in model.parameters() if p.requires_grad)
fig = bar_plot(['All parameters','Trainable head'],[total_count,trainable_count],'Declared parameter update policy','Scalar parameters')

Read the result

The trainable count is much smaller than the full count, but the backbone remains in memory and still computes features. Its initialization provenance remains unchanged.

Change and compare

  • Unfreeze only final_norm in a separate copy and recount.
  • Explain why model.train() would not unfreeze a parameter.

Guided task 05

Cross entropy uses one class target per row

Classification cross entropy compares one raw class-logit vector with one integer class target. Unlike language-model training, there is no next-token shift and no vocabulary-sized target at each token position. Applying a language-model flattening recipe without checking these roles would change the task or produce a shape error.

For each row, log_softmax creates stable log probabilities, and the loss selects the negative value at the target class. The default mean averages over examples. Inspect individual losses before aggregating so a large error is not hidden by a single batch mean. Different messages may have different token lengths, but this classification objective gives each message one contribution rather than weighting long messages by their token count. During full evaluation with uneven batches, accumulate summed losses and divide by the total number of examples. The current loss measures a newly initialized head, not a meaningful trained baseline for service deployment. Its purpose is to verify that class order, tensor dimensions and the mathematical objective agree.

Try it

  1. Recompute raw logits under the declared head.
  2. Calculate per-example cross entropy and a manual log-softmax equivalent.
  3. Plot individual contributions before taking the mean.
Python · run in order
logits = model(ids,valid)
individual_loss = F.cross_entropy(logits,labels,reduction='none')
manual_loss = -F.log_softmax(logits,dim=1)[torch.arange(len(labels)),labels]
loss = individual_loss.mean()
fig = bar_plot([r['id'] for r in rows],individual_loss.detach().numpy(),'Classification loss contributions','Negative log probability')

Read the result

Each message contributes one loss regardless of how much padding surrounds it. Its class target selects a score coordinate, not a token position.

Change and compare

  • Check the loss for logits [0,0] and either class label.
  • Explain why passing softmax probabilities into cross entropy changes the calculation.

Guided task 06

Verify the update boundary with gradients

An optimizer step should change exactly the parameter groups allowed by the chosen policy. Capture the backbone and head before computing loss, clear old gradients, backpropagate once, and inspect the resulting gradient norms. A nonzero head gradient shows a learning signal for this batch; it does not promise that the update improves unseen requests.

The backbone is frozen, so its parameters should have no accumulated gradients and should remain bitwise unchanged after the head update. Do not infer this only from the optimizer constructor: flags, stale gradients and later code changes can invalidate an intended policy. Explicit comparisons make the boundary testable. This task keeps the whole model in evaluation mode so frozen features have no random dropout changes, while the trainable head still participates in autograd. Evaluation mode does not forbid optimization. The separate concepts are layer behavior, graph recording, parameter gradient flags and optimizer membership. Keeping them distinct is especially useful when adapting only a small subset of a larger pretrained network.

Try it

  1. Snapshot backbone and head weights before an update.
  2. Backpropagate head-only cross entropy and measure the head gradient norm.
  3. Apply one optimizer step and verify the frozen state.
Python · run in order
backbone_before = {k:v.detach().clone() for k,v in model.backbone.state_dict().items()}
head_before = model.head.weight.detach().clone()
optimizer = torch.optim.SGD([p for p in model.parameters() if p.requires_grad],lr=0.05)
optimizer.zero_grad(set_to_none=True)
update_loss = F.cross_entropy(model(ids,valid),labels)
update_loss.backward()
head_gradient = float(model.head.weight.grad.norm())
optimizer.step()
head_change = float((model.head.weight.detach()-head_before).norm())
backbone_change = sum(float((v-backbone_before[k]).abs().sum()) for k,v in model.backbone.state_dict().items())
fig = bar_plot(['Head gradient norm','Head weight change','Backbone absolute change'],[head_gradient,head_change,backbone_change],'One measured head-only update','Norm or absolute sum')

Read the result

The zero backbone change checks the intended frozen boundary. The head movement verifies an update happened, without equating one batch step with better classification.

Change and compare

  • Repeat with learning rate zero and separate gradient existence from parameter movement.
  • Explain why old gradients should be cleared before a new batch.

Guided task 07

Test invariance to extra right padding

Adding unused right-padding columns should not change a length-aware classifier's prediction on the same real message under evaluation mode. The real token positions, their positional embeddings and their available causal context are unchanged. Extra padding keys are excluded, and final pooling still uses the original last valid coordinate.

This is a useful implementation property to test directly. It can expose missing key-padding masks, blind final-column pooling, or a mistaken length calculation. Use the same weights and deterministic mode in both calls so dropout randomness does not obscure the comparison. Compare logits with a small floating tolerance rather than requiring a changed kernel layout to produce bit-identical arithmetic. Passing this test proves a narrow input-invariance property, not semantic correctness or general classification quality. It also does not establish invariance to left padding, which can change absolute position IDs unless handled separately. The stated contract remains nonempty, right-padded sequences within the model's context capacity.

Try it

  1. Append three PAD columns and three false mask columns.
  2. Evaluate original and extended batches without gradients.
  3. Plot the maximum absolute logit difference per message.
Python · run in order
extra_ids = F.pad(ids,(0,3),value=0)
extra_valid = F.pad(valid,(0,3),value=False)
model.eval()
with torch.no_grad():
    normal_logits = model(ids,valid)
    padded_logits = model(extra_ids,extra_valid)
padding_error = (normal_logits-padded_logits).abs().amax(1)
fig = bar_plot([r['id'] for r in rows],padding_error.numpy(),'Extra right-padding invariance check','Maximum absolute logit difference')

Read the result

The plotted differences should be near numerical precision. A large discrepancy calls for inspecting masking and pooling before discussing model quality.

Change and compare

  • Test two different amounts of extra right padding.
  • Change only a real token and explain why invariance should no longer be expected.

Guided task 08

Evaluation mode is not gradient disabling

Evaluation mode changes the behavior of dropout in the backbone, while no_grad controls whether PyTorch records operations for differentiation. Neither command substitutes for the other. A model in evaluation mode can still produce a differentiable output when its head requires gradients; a model in training mode can still apply dropout inside a no_grad block.

Make both choices explicit at inference. Compare two evaluation calls under no_grad and inspect whether a separate evaluation-mode output tracks gradients. Then briefly switch to training mode inside no_grad and measure the difference between two dropout-affected outputs. Return to evaluation mode afterward so the lesson leaves a predictable state. A fixed seed defines a random sequence, not a guarantee that successive dropout masks are identical. The core model remains a small randomly initialized teaching backbone with one head update. Its numerical checks prepare you to use a real pretrained backbone responsibly, but no language-understanding or deployment-accuracy claim follows from them.

Try it

  1. Inspect requires_grad on an evaluation-mode output.
  2. Compare repeated no-grad evaluation calls.
  3. Contrast training-mode dropout behavior and restore evaluation mode.
Python · run in order
model.eval()
tracked = model(ids,valid)
with torch.no_grad():
    eval_a = model(ids,valid)
    eval_b = model(ids,valid)
model.train()
with torch.no_grad():
    train_a = model(ids,valid)
    train_b = model(ids,valid)
model.eval()
differences = [float((eval_a-eval_b).abs().max()),float((train_a-train_b).abs().max())]
fig = bar_plot(['Evaluation calls','Training-mode calls'],differences,'Measured repeat-call differences','Maximum absolute logit difference')

Read the result

A graph-tracked evaluation output and a dropout-affected no-grad output can both be correct under their respective settings. State both controls in an inference recipe.

Change and compare

  • Check that all parameter flags are unchanged by train and eval.
  • Explain why repeated stochastic differences are not a confidence estimate.

Extension 1

Optional real pretrained GPT-2 backbone

This standalone route downloads approximately 500 MB of pretrained weights on first use. Allow at least 8 GB host RAM; a GPU is recommended for longer runs. The GPT-2 backbone is pretrained but its two-class head is new. This short run demonstrates mechanics on constructed requests, not a reliable service classifier. Pool using attention-mask lengths so genuine EOS and EOS padding remain distinct. Training and held-out requests are separate; no paid API is used.

Python · standalone extension
import torch
from transformers import GPT2TokenizerFast, GPT2ForSequenceClassification
torch.manual_seed(316)
torch.set_num_threads(1)
checkpoint = 'openai-community/gpt2'
tokenizer = GPT2TokenizerFast.from_pretrained(checkpoint, revision='607a30d783dfa663caf39e06633721c8d4cfcd7e')
tokenizer.pad_token = tokenizer.eos_token
tokenizer.padding_side = 'right'
model = GPT2ForSequenceClassification.from_pretrained(checkpoint, revision='607a30d783dfa663caf39e06633721c8d4cfcd7e', use_safetensors=True, num_labels=2, pad_token_id=tokenizer.eos_token_id)
device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
model.to(device)
training_texts = ['reserve a room today', 'please book a study room', 'cancel my room request', 'remove my room booking']
heldout_texts = ['reserve a room for tomorrow', 'cancel the room for tomorrow']
training_labels = torch.tensor([0,0,1,1], device=device)
heldout_labels = torch.tensor([0,1], device=device)
def encode_messages(texts):
    batch = tokenizer(texts, padding=True, truncation=True, max_length=48, return_tensors='pt')
    return {key:value.to(device) for key,value in batch.items()}
train_batch = encode_messages(training_texts)
heldout_batch = encode_messages(heldout_texts)
def logits_for(batch):
    hidden = model.transformer(input_ids=batch['input_ids'], attention_mask=batch['attention_mask'], use_cache=False).last_hidden_state
    last = batch['attention_mask'].sum(1)-1
    return model.score(hidden)[torch.arange(len(last),device=device), last]
for parameter in model.parameters():
    parameter.requires_grad = False
for parameter in model.score.parameters():
    parameter.requires_grad = True
model.eval()  # Frozen backbone dropout is disabled while fitting the head.
with torch.no_grad():
    before_logits = logits_for(heldout_batch)
    before_loss = torch.nn.functional.cross_entropy(before_logits,heldout_labels).item()
    before_accuracy = (before_logits.argmax(1)==heldout_labels).float().mean().item()
optimizer = torch.optim.AdamW(model.score.parameters(),lr=0.001)
for step in range(1):
    optimizer.zero_grad(set_to_none=True)
    loss = torch.nn.functional.cross_entropy(logits_for(train_batch),training_labels)
    loss.backward()
    optimizer.step()
with torch.no_grad():
    after_logits = logits_for(heldout_batch)
    after_loss = torch.nn.functional.cross_entropy(after_logits,heldout_labels).item()
    after_accuracy = (after_logits.argmax(1)==heldout_labels).float().mean().item()
print({'heldout_before':{'loss':before_loss,'accuracy':before_accuracy}, 'heldout_after':{'loss':after_loss,'accuracy':after_accuracy}, 'trainable_parameters':sum(p.numel() for p in model.parameters() if p.requires_grad)})
