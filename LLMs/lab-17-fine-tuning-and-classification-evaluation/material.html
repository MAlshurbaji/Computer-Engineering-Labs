Fine-Tuning and Classification Evaluation

MaterialPracticeQuiz

Run a small classification experiment whose data roles stay fixed from start to finish. Compare the same held-out requests before and after updates, and keep the result tied to its actual evidence.

What you will learn

  • Measure a held-out baseline before optimization.
  • Train with explicit parameter and data policies.
  • Aggregate metrics over all intended examples.
  • Preserve classification and preprocessing state for inference.

Before this lab: Use Python, token IDs, tensor axes and the causal-transformer concepts from preceding labs.

In this lab

  1. Record a paired held-out baseline
  2. Declare the update budget before training
  3. Train only on the training requests
  4. Weight evaluation by examples, not batches
  5. A confusion matrix preserves error direction
  6. Inspect confidence and mistakes separately
  7. Use final test data only after fixing the run
  8. Save the model with its input meanings

Python · shared setup
import copy, json, re, tempfile
from pathlib import Path
import numpy as np
import torch
from torch import nn
from torch.nn import functional as F
import matplotlib.pyplot as plt
torch.set_num_threads(1)
torch.manual_seed(117)
# Each pair describes the same room request; group before splitting.
pairs = [
    (0, ('reserve a quiet room', 'please reserve the quiet room')),
    (1, ('cancel my quiet room', 'please cancel the quiet room')),
    (0, ('reserve a large room', 'please reserve the large room')),
    (1, ('cancel my large room', 'please cancel the large room')),
    (0, ('reserve a room today', 'please reserve my room today')),
    (1, ('cancel a room today', 'please cancel my room today')),
    (0, ('reserve the room tomorrow', 'please reserve a room tomorrow')),
    (1, ('cancel the room tomorrow', 'please cancel a room tomorrow')),
    (0, ('reserve a sunny room', 'please reserve the sunny room')),
    (1, ('cancel my sunny room', 'please cancel the sunny room')),
    (0, ('reserve a corner room', 'please reserve the corner room')),
    (1, ('cancel my corner room', 'please cancel the corner room')),
]
records = [dict(id=f'g{g}_v{v}', group=g, text=text, label=label)
           for g, (label, texts) in enumerate(pairs) for v, text in enumerate(texts)]
split_groups = {'train': set(range(8)), 'validation': {8, 9}, 'test': {10, 11}}
def words(text):
    return re.findall(r"[a-z]+", text.lower())
# Vocabulary is fitted on training records only; IDs 0/1 mean PAD/UNK.
training_words = sorted({word for r in records if r['group'] in split_groups['train'] for word in words(r['text'])})
vocab = {'<pad>': 0, '<unk>': 1, **{word: i+2 for i, word in enumerate(training_words)}}
def encode(text, limit=12):
    ids = [vocab.get(word, 1) for word in words(text)][:limit]
    if not ids:
        raise ValueError('A message must contain at least one token.')
    return ids

def make_batch(rows, limit=12):
    if not rows:
        raise ValueError('Cannot collate an empty batch.')
    sequences = [encode(r['text'], limit) for r in rows]
    lengths = torch.tensor([len(s) for s in sequences], dtype=torch.long)
    ids = torch.zeros((len(rows), int(lengths.max())), dtype=torch.long)
    for i, sequence in enumerate(sequences):
        ids[i, :len(sequence)] = torch.tensor(sequence)
    valid = torch.arange(ids.shape[1])[None, :] < lengths[:, None]
    labels = torch.tensor([r['label'] for r in rows], dtype=torch.long)
    return ids, valid, labels
# One small causal transformer block, randomly initialized: not pretrained.
class TinyBackbone(nn.Module):
    def __init__(self, vocabulary_size, width=24, context=32):
        super().__init__()
        self.token = nn.Embedding(vocabulary_size, width)
        self.position = nn.Embedding(context, width)
        self.norm1 = nn.LayerNorm(width)
        self.attention = nn.MultiheadAttention(width, 4, dropout=0.1, batch_first=True)
        self.norm2 = nn.LayerNorm(width)
        self.ff = nn.Sequential(nn.Linear(width, width*2), nn.GELU(), nn.Linear(width*2, width))
        self.final_norm = nn.LayerNorm(width)
    def forward(self, ids, valid):
        if ids.ndim != 2 or ids.shape != valid.shape or ids.shape[1] > self.position.num_embeddings:
            raise ValueError('Invalid token shape or context length.')
        if not torch.all(valid.any(dim=1)):
            raise ValueError('Every row needs a real token.')
        length = ids.shape[1]
        x = self.token(ids) + self.position(torch.arange(length, device=ids.device))[None]
        causal = torch.triu(torch.ones(length, length, dtype=torch.bool, device=ids.device), diagonal=1)
        normalized = self.norm1(x)
        attended, _ = self.attention(normalized, normalized, normalized, attn_mask=causal, key_padding_mask=~valid, need_weights=False)
        x = x + attended
        return self.final_norm(x + self.ff(self.norm2(x)))

class TinyClassifier(nn.Module):
    def __init__(self):
        super().__init__()
        self.backbone = TinyBackbone(len(vocab))
        self.head = nn.Linear(24, 2)
    def forward(self, ids, valid):
        hidden = self.backbone(ids, valid)
        last = valid.sum(dim=1) - 1  # Right padding, never select a padded position.
        pooled = hidden[torch.arange(ids.shape[0], device=ids.device), last]
        return self.head(pooled)
# Example-weighted evaluation covers the full declared record list.
def evaluate_classifier(model, rows, batch_size=3):
    if not rows or batch_size < 1:
        raise ValueError('Require examples and positive batch size.')
    model.eval()
    loss_sum, count, correct = 0.0, 0, 0
    all_logits, all_labels = [], []
    with torch.no_grad():
        for start in range(0,len(rows),batch_size):
            ids,valid,labels = make_batch(rows[start:start+batch_size])
            logits = model(ids,valid)
            loss_sum += float(F.cross_entropy(logits,labels,reduction='sum'))
            count += len(labels)
            correct += int((logits.argmax(1)==labels).sum())
            all_logits.append(logits)
            all_labels.append(labels)
    return {'loss':loss_sum/count,'accuracy':correct/count,'count':count,
            'logits':torch.cat(all_logits),'labels':torch.cat(all_labels)}
# Small plotting helpers display measured tensors and counts.
def bar_plot(labels, values, title, ylabel):
    fig, ax = plt.subplots(figsize=(7, 3.5))
    ax.bar(labels, np.asarray(values))
    ax.set(title=title, ylabel=ylabel)
    if np.all(np.asarray(values)==0):
        ax.set_ylim(0,1)
        ax.text(0.5,0.5,'All values are zero',transform=ax.transAxes,ha='center')
    fig.tight_layout()
    return fig

def heat_plot(values, title, xlabel, ylabel, label):
    array = values.detach().cpu().numpy() if isinstance(values, torch.Tensor) else np.asarray(values)
    fig, ax = plt.subplots(figsize=(7, 3.5))
    if np.any(array < 0):
        bound = max(float(np.abs(array).max()), 1e-8)
        view = ax.imshow(array, aspect='auto', interpolation='nearest', cmap='coolwarm', vmin=-bound, vmax=bound)
    else:
        view = ax.imshow(array, aspect='auto', interpolation='nearest', cmap='viridis', vmin=0, vmax=max(float(array.max()),1e-8))
    ax.set_xticks(np.unique(np.linspace(0,array.shape[1]-1,min(8,array.shape[1])).astype(int)))
    ax.set_yticks(np.unique(np.linspace(0,array.shape[0]-1,min(8,array.shape[0])).astype(int)))
    ax.set(title=title, xlabel=xlabel, ylabel=ylabel)
    fig.colorbar(view, ax=ax, label=label)
    fig.tight_layout()
    return fig

Guided task 01

Record a paired held-out baseline

An adaptation experiment needs a baseline measured on the same held-out examples used after training. Create a fresh tiny classifier and record its validation logits, loss and accuracy before any updates. The baseline here is randomly initialized, so the core exercise demonstrates training and evaluation mechanics rather than transfer from pretrained language knowledge. The optional extension uses a real pretrained GPT-2 backbone with a new task head.

Keep a separate baseline model copy so its parameters cannot change while the candidate trains. Both versions use the same training-fitted vocabulary, right-padding mask and last-valid-token pooling. A comparison with different tokenization, examples or class mappings would combine several changes and become harder to interpret. Validation examples support development checks; final test requests remain unused until the configuration is fixed. The four validation messages are related in two groups and far too few to support broad reliability claims. Their measured results are still useful for checking that the comparison procedure and metric accounting are implemented correctly.

Try it

  1. Collect fixed training and validation rows by group.
  2. Create an untouched baseline and a separate candidate model.
  3. Measure baseline probabilities on every validation request.
Python · run in order
train_rows = [r for r in records if r['group'] in split_groups['train']]
validation_rows = [r for r in records if r['group'] in split_groups['validation']]

Guided task 02

Declare the update budget before training

A reproducible training run states which parameters can change, which records provide gradients, and how many updates are allowed. This core run trains all parameters of the tiny model for forty full-batch steps with AdamW. It is deliberately small enough to inspect on CPU. The pretrained extension updates the final transformer block, final normalization and classification head while earlier layers remain frozen.

Construct the optimizer after selecting trainable parameters. Learning rate, weight decay and step count are part of the experiment configuration, not hidden defaults to adjust after viewing final test results. Full-batch training here keeps the example code compact; it does not imply that larger datasets should fit into one batch. A real minibatch loop also needs a declared presentation order and aggregation policy. The model includes dropout, so training mode is re-entered before each update and evaluation uses deterministic mode. Store the configuration as plain data now, before generating a favorable-looking curve. This makes an unchanged failed experiment just as reportable as one with lower validation loss.

Try it

  1. Set forty steps, learning rate 0.01 and weight decay 0.01.
  2. Confirm that all tiny-model parameters are trainable.
  3. Count updated parameters and record the configuration.
Python · run in order
configuration = {'steps':40,'learning_rate':0.01,'weight_decay':0.01,'update_policy':'all tiny-model parameters','seed':117}
for p in candidate.parameters():
    p.requires_grad=True
optimizer = torch.optim.AdamW(candidate.parameters(),lr=configuration['learning_rate'],weight_decay=configuration['weight_decay'])
trainable = sum(p.numel() for p in candidate.parameters() if p.requires_grad)
frozen = sum(p.numel() for p in candidate.parameters() if not p.requires_grad)
fig = bar_plot(['Trainable','Frozen'],[trainable,frozen],'Core run parameter policy','Scalar parameters')

Guided task 03

Train only on the training requests

The training loop clears old gradients, computes class logits, calculates cross entropy, backpropagates and takes one optimizer step. Unlike an autoregressive objective, the target is a single class ID per message, so there is no token-label shift. The attention mask and length-aware pooling remain active during training as well as evaluation.

Record the actual loss before each update. Dropout and changing parameters can make the curve fluctuate; do not require every step to decrease. Every tenth step, evaluate the full validation collection with gradients disabled, then explicitly return to training mode on the next update. Evaluation must not call optimizer.step or modify the labels. The optimizer never receives validation examples for gradient computation. A small constructed dataset can be memorized quickly, particularly when action words repeat. This makes the run useful for verifying that learning signals flow through the implementation, but it does not establish general language ability or robustness to ambiguous real requests. Keep all measured outcomes, including runs whose held-out metrics do not improve.

Try it

  1. Create one full training batch with the fixed collator.
  2. Run the declared optimizer loop and retain each measured loss.
  3. Measure validation only at the stated checkpoints.
Python · run in order
train_ids,train_valid,train_labels = make_batch(train_rows)
training_losses,validation_losses,check_steps = [],[],[]
for step in range(configuration['steps']):
    candidate.train()
    optimizer.zero_grad(set_to_none=True)
    loss = F.cross_entropy(candidate(train_ids,train_valid),train_labels)
    loss.backward()
    optimizer.step()
    training_losses.append(float(loss.detach()))
    if (step+1)%10==0:
        validation_losses.append(evaluate_classifier(candidate,validation_rows)['loss'])
        check_steps.append(step+1)
fig,ax = plt.subplots(figsize=(7,3.5))
ax.plot(np.arange(1,41),training_losses,label='Training batch loss before update')
ax.plot(check_steps,validation_losses,'o-',label='Full validation loss after update')
ax.set(xlabel='Optimizer step',ylabel='Cross entropy',title='Measured training and validation trajectories')
ax.legend()
fig.tight_layout()

Guided task 04

Weight evaluation by examples, not batches

A loader can end with a smaller batch. Averaging its batch-mean losses equally would give each example in the short batch too much influence. Instead sum all per-example losses and divide by the total number of examples. Classification accuracy similarly counts correct predictions over the complete subset. This is different from token-weighted language-model evaluation, where the denominator is the number of supervised target tokens.

The evaluation helper follows this rule and includes the remainder batch. Check its result using batch sizes one, three and four on the same validation records. Minor floating differences may occur from accumulation order, but the intended statistic should agree. The helper also returns logits and labels in the original record order for later error analysis. It does not shuffle, train or silently discard examples. An empty evaluation subset is rejected rather than producing a misleading zero or a division error. These accounting choices become especially important when classes are rare or the dataset is small, because losing or over-weighting one example can substantially change the reported result.

Try it

  1. Evaluate the same candidate using three batch sizes.
  2. Compare full-subset loss and accuracy.
  3. Verify the total consumed-example count.

Read the result

Guided task 05

A confusion matrix preserves error direction

Accuracy compresses every error into one count, while a confusion matrix keeps the relationship between true and predicted classes. Rows here are true actions and columns are predicted actions. An off-diagonal count therefore distinguishes a reservation wrongly canceled from a cancellation wrongly reserved. State that orientation explicitly because reversing it changes the interpretation of each cell.

Build the matrix from the actual held-out predictions rather than entering illustrative counts as if they were measured. Its total must equal the number of evaluated messages, and row totals must equal class support. In this tiny fixture each class has only two validation examples, so percentages would look more precise than the evidence deserves. Show integer counts and retain the underlying record IDs for review. A perfect matrix, if this run produces one, still would not establish competence on unfamiliar vocabulary, negation or multiple-action requests. The result is a local measurement under the fixed contract. Use it to identify which type of mistake occurred, not to infer the cause without inspecting examples.

Try it

  1. Take predictions from the full validation evaluation.
  2. Accumulate counts with true labels on rows.
  3. Display measured counts and their orientation clearly.
Python · run in order
predictions = after['logits'].argmax(1)
confusion = torch.zeros((2,2),dtype=torch.long)
for true,predicted in zip(after['labels'],predictions):
    confusion[true,predicted]+=1
fig,ax = plt.subplots(figsize=(5,4))
im=ax.imshow(confusion.numpy(),cmap='Blues',vmin=0)
for row in range(2):
    for col in range(2):
        value=int(confusion[row,col])
        ax.text(col,row,str(value),ha='center',va='center',color='white' if value>float(confusion.max())/2 else 'black')
ax.set(xticks=[0,1],yticks=[0,1],xticklabels=['reserve','cancel'],yticklabels=['reserve','cancel'],xlabel='Predicted action',ylabel='True action',title='Measured validation confusion counts')
fig.colorbar(im,ax=ax,label='Messages')
fig.tight_layout()

Guided task 06

Inspect confidence and mistakes separately

The largest softmax value is a model score, not a guarantee that the selected action is correct. Pair it with the true label, predicted label and record identity before discussing confidence. A confidently wrong cancellation could be more consequential than an uncertain reservation, but a decision policy needs evidence about the application's costs rather than an arbitrary threshold chosen to flatter the plot.

The task compares baseline and adapted correctness on exactly the same validation records. Some records can improve while others worsen even when an aggregate metric looks better. Keep both directions visible. A tiny validation set cannot support a dependable calibration analysis, so the lesson does not turn these values into claims about real-world error probabilities. Reading the text can reveal obvious limits of the toy setup, such as its repeated action words and sparse phrasing. Use those observations to propose future data collection or a controlled experiment. Do not remove inconvenient held-out examples after seeing their predictions, and do not reassign their labels simply because the model disagrees.

Guided task 07

Use final test data only after fixing the run

The final test collection answers a different question from the validation checks used during development. Here the update policy, step count and preprocessing were fixed before test evaluation. Evaluate the baseline and candidate on the same final test records once, then report both outcomes without changing settings to improve that comparison.

Paired baseline and adapted test measurements are legitimate when the comparison was planned in advance; repeatedly choosing a new configuration after inspecting test results would turn the test set into another development set. Preserve the number of messages and request groups alongside the metrics. Our final test subset contains only two underlying requests with two phrasings each, so its uncertainty and narrow coverage remain substantial. A favorable result cannot support a claim of robust room-booking automation. A disappointing result should remain part of the report rather than being replaced with a different split. The purpose of this task is to complete the declared evaluation procedure and leave a clear distinction between observed toy performance and future deployment requirements.

Guided task 08

Save the model with its input meanings

A classification checkpoint needs more than a tensor of head weights. The consumer must know the backbone architecture, class mapping, vocabulary, tokenization rule, padding convention and maximum length. Save the model state with that configuration, plus optimizer state and progress when retaining a training checkpoint. Use a fresh temporary directory so the exercise does not overwrite another run.

Reload into a newly constructed compatible model and compare deterministic logits on the same input. Equality of predictions alone would be a weaker check because different logits can share an argmax. Loading state dictionaries with weights_only=True keeps the example within a plain-data checkpoint format. The saved RNG tensor and optimizer state support future continuation work, but this task verifies inference replay rather than claiming an exact interrupted-training replay across devices or library versions. Record the tiny random-initialization provenance and the fixed data roles. The pretrained extension requires its real checkpoint identity as well; a toy architecture cannot accept those weights simply because both models use the name GPT.

Try it

  2. Load a new compatible classifier with weights_only=True.
  3. Compare reloaded and original evaluation logits.

Extension 1

Optional real pretrained classification fine-tuning

This standalone route downloads approximately 500 MB of pretrained weights on first use. Allow at least 8 GB host RAM; a GPU is recommended for longer runs. The GPT-2 backbone is pretrained but its two-class head is new. This short run demonstrates mechanics on constructed requests, not a reliable service classifier. Pool using attention-mask lengths so genuine EOS and EOS padding remain distinct. Training and held-out requests are separate; no paid API is used. This extension adapts the final transformer block, final layer normalization and head; earlier backbone layers stay frozen. Eight short updates are a mechanics demonstration, not a performance guarantee.

In the checked CPU run, held-out loss decreased while accuracy on the two examples stayed unchanged. Inspect both measurements before drawing a conclusion.

Python · standalone extension
import torch
from transformers import GPT2TokenizerFast, GPT2ForSequenceClassification
torch.manual_seed(316)
torch.set_num_threads(1)
checkpoint = 'openai-community/gpt2'
tokenizer = GPT2TokenizerFast.from_pretrained(checkpoint, revision='607a30d783dfa663caf39e06633721c8d4cfcd7e')
tokenizer.pad_token = tokenizer.eos_token
tokenizer.padding_side = 'right'
model = GPT2ForSequenceClassification.from_pretrained(checkpoint, revision='607a30d783dfa663caf39e06633721c8d4cfcd7e', use_safetensors=True, num_labels=2, pad_token_id=tokenizer.eos_token_id)
device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
model.to(device)
training_texts = ['reserve a room today', 'please book a study room', 'cancel my room request', 'remove my room booking']
heldout_texts = ['reserve a room for tomorrow', 'cancel the room for tomorrow']
training_labels = torch.tensor([0,0,1,1], device=device)
heldout_labels = torch.tensor([0,1], device=device)
def encode_messages(texts):
    batch = tokenizer(texts, padding=True, truncation=True, max_length=48, return_tensors='pt')
    return {key:value.to(device) for key,value in batch.items()}
train_batch = encode_messages(training_texts)
heldout_batch = encode_messages(heldout_texts)
def logits_for(batch):
    hidden = model.transformer(input_ids=batch['input_ids'], attention_mask=batch['attention_mask'], use_cache=False).last_hidden_state
    last = batch['attention_mask'].sum(1)-1
    return model.score(hidden)[torch.arange(len(last),device=device), last]
for parameter in model.parameters():
    parameter.requires_grad = False
for parameter in model.score.parameters():
    parameter.requires_grad = True
for module in [model.transformer.h[-1], model.transformer.ln_f]:
    for parameter in module.parameters():
        parameter.requires_grad = True
model.eval()
with torch.no_grad():
    before_logits = logits_for(heldout_batch)
    before_loss = torch.nn.functional.cross_entropy(before_logits,heldout_labels).item()
    before_accuracy = (before_logits.argmax(1)==heldout_labels).float().mean().item()
optimizer = torch.optim.AdamW([p for p in model.parameters() if p.requires_grad],lr=0.00005)
for step in range(8):
    model.train()
    optimizer.zero_grad(set_to_none=True)
    loss = torch.nn.functional.cross_entropy(logits_for(train_batch),training_labels)
    loss.backward()
    optimizer.step()
model.eval()
with torch.no_grad():
    after_logits = logits_for(heldout_batch)
    after_loss = torch.nn.functional.cross_entropy(after_logits,heldout_labels).item()
    after_accuracy = (after_logits.argmax(1)==heldout_labels).float().mean().item()
print({'heldout_before':{'loss':before_loss,'accuracy':before_accuracy}, 'heldout_after':{'loss':after_loss,'accuracy':after_accuracy}, 'trainable_parameters':sum(p.numel() for p in model.parameters() if p.requires_grad)})
