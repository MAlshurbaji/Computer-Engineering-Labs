Response Evaluation and Final Project

MaterialPracticeQuiz

Bring the training and evaluation pieces together in a small, inspectable project. You will preserve actual successes and failures, apply several kinds of evidence, and verify a saved model without overstating what it can do.

What you will learn

  • Define evaluation rules before inspecting final outcomes.
  • Separate generated response tokens from the prompt.
  • Combine task checks with a clearly unscored human rubric.
  • Replay an artifact containing model state and evaluation context.

Before this lab: Use Python, token IDs, tensor axes and the causal-transformer concepts from preceding labs.

import copy, json, re, tempfile
from pathlib import Path
import numpy as np
import torch
from torch import nn
from torch.nn import functional as F
import matplotlib.pyplot as plt
torch.set_num_threads(1)
torch.manual_seed(120)
# Fixed character vocabulary: never fitted on held-out responses.
characters = '\n' + ''.join(chr(i) for i in range(32,127))
char_to_id = {character:i+3 for i,character in enumerate(characters)}
id_to_char = {value:key for key,value in char_to_id.items()}
BOS, EOS, PAD = 1, 2, 2
VOCAB_SIZE = len(char_to_id)+3
instruction_rows = [
    {'id':'train0','split':'train','input':'library','response':'bring a book'},
    {'id':'train1','split':'train','input':'park','response':'bring water'},
    {'id':'train2','split':'train','input':'rain','response':'bring an umbrella'},
    {'id':'train3','split':'train','input':'gym','response':'bring a towel'},
    {'id':'train4','split':'train','input':'school','response':'bring a book'},
    {'id':'train5','split':'train','input':'picnic','response':'bring water'},
    {'id':'train6','split':'train','input':'showers','response':'bring an umbrella'},
    {'id':'train7','split':'train','input':'swimming','response':'bring a towel'},
    {'id':'val0','split':'validation','input':'reading cafe','response':'bring a book'},
    {'id':'val1','split':'validation','input':'workout','response':'bring a towel'},
    {'id':'test0','split':'test','input':'hot day','response':'bring water'},
    {'id':'test1','split':'test','input':'wet weather','response':'bring an umbrella'},
]
for row in instruction_rows:
    row['instruction'] = 'name one useful item'
    row['group'] = row['id']  # Each supplied situation has one record.

def encode_text(text):
    if any(character not in char_to_id for character in text):
        raise ValueError('Only the declared character vocabulary is supported.')
    return [char_to_id[character] for character in text]

def format_prompt(row):
    return 'Task: '+row['instruction']+'\nInput: '+row['input']+'\nReply: '

def instruction_parts(row):
    response = row['response'].strip()
    if not response:
        raise ValueError('A response must contain visible text.')
    prefix = [BOS]+encode_text(format_prompt(row))
    target = encode_text(response)+[EOS]
    return prefix,target

def pack_instructions(rows,max_length=128):
    if not rows:
        raise ValueError('Cannot collate an empty batch.')
    parts = [instruction_parts(row) for row in rows]
    lengths = [len(prefix)+len(target) for prefix,target in parts]
    if any(length>max_length for length in lengths):
        raise ValueError('Complete response and EOS must fit; shorten or reject the record.')
    ids = torch.full((len(rows),max(lengths)),PAD,dtype=torch.long)
    valid = torch.zeros_like(ids,dtype=torch.bool)
    labels = torch.full_like(ids,-100)
    for i,(prefix,target) in enumerate(parts):
        sequence = prefix+target
        ids[i,:len(sequence)] = torch.tensor(sequence)
        valid[i,:len(sequence)] = True
        labels[i,len(prefix):len(sequence)] = torch.tensor(target)
    return ids,valid,labels
# One small causal transformer block, randomly initialized: not pretrained.
class TinyBackbone(nn.Module):
    def __init__(self, vocabulary_size, width=24, context=32):
        super().__init__()
        self.token = nn.Embedding(vocabulary_size, width)
        self.position = nn.Embedding(context, width)
        self.norm1 = nn.LayerNorm(width)
        self.attention = nn.MultiheadAttention(width, 4, dropout=0.1, batch_first=True)
        self.norm2 = nn.LayerNorm(width)
        self.ff = nn.Sequential(nn.Linear(width, width*2), nn.GELU(), nn.Linear(width*2, width))
        self.final_norm = nn.LayerNorm(width)
    def forward(self, ids, valid):
        if ids.ndim != 2 or ids.shape != valid.shape or ids.shape[1] > self.position.num_embeddings:
            raise ValueError('Invalid token shape or context length.')
        if not torch.all(valid.any(dim=1)):
            raise ValueError('Every row needs a real token.')
        length = ids.shape[1]
        x = self.token(ids) + self.position(torch.arange(length, device=ids.device))[None]
        causal = torch.triu(torch.ones(length, length, dtype=torch.bool, device=ids.device), diagonal=1)
        normalized = self.norm1(x)
        attended, _ = self.attention(normalized, normalized, normalized, attn_mask=causal, key_padding_mask=~valid, need_weights=False)
        x = x + attended
        return self.final_norm(x + self.ff(self.norm2(x)))

class TinyInstructionLM(nn.Module):
    def __init__(self):
        super().__init__()
        self.backbone = TinyBackbone(VOCAB_SIZE,width=24,context=128)
        self.output = nn.Linear(24,VOCAB_SIZE,bias=False)
        self.output.weight = self.backbone.token.weight
    def forward(self,ids,valid):
        return self.output(self.backbone(ids,valid))

def response_loss_sum(logits,labels):
    targets = labels[:,1:]  # The local model does not shift inside forward.
    count = int((targets!=-100).sum())
    if count==0:
        raise ValueError('No supervised next-token targets.')
    total = F.cross_entropy(logits[:,:-1].reshape(-1,VOCAB_SIZE),targets.reshape(-1),ignore_index=-100,reduction='sum')
    return total,count

def evaluate_instructions(model,rows,batch_size=2):
    if not rows or batch_size<1:
        raise ValueError('Need records and a positive batch size.')
    model.eval()
    total,count = 0.0,0
    with torch.no_grad():
        for start in range(0,len(rows),batch_size):
            ids,valid,labels = pack_instructions(rows[start:start+batch_size])
            loss,n = response_loss_sum(model(ids,valid),labels)
            total += float(loss)
            count += n
    return {'loss':total/count,'tokens':count,'records':len(rows)}

def generate_reply(model,row,max_new_tokens=24):
    if max_new_tokens<1:
        raise ValueError('Generation budget must be positive.')
    sequence = [BOS]+encode_text(format_prompt(row))
    if len(sequence)>=128:
        raise ValueError('The prompt leaves no generation space.')
    generated = []
    reason = 'budget'
    model.eval()
    with torch.no_grad():
        for _ in range(max_new_tokens):
            if len(sequence)>=128:
                reason='context'
                break
            ids = torch.tensor([sequence])
            scores = model(ids,torch.ones_like(ids,dtype=torch.bool))[0,-1].clone()
            scores[0] = scores[BOS] = -torch.inf
            next_id = int(scores.argmax())
            sequence.append(next_id)
            generated.append(next_id)
            if next_id==EOS:
                reason='eos'
                break
    text = ''.join(id_to_char[token] for token in generated if token in id_to_char)
    return {'text':text,'token_ids':generated,'stop':reason}
def normalize_reply(text):
    return ' '.join(text.lower().split())

def response_checks(text,reference):
    normalized=normalize_reply(text)
    words=normalized.split()
    format_ok=2<=len(words)<=4 and words[0]=='bring' and all(word.isalpha() and word.isascii() for word in words)
    return {'exact_match':normalized==normalize_reply(reference),'format_ok':bool(format_ok)}

def evaluate_responses(model,rows,budget=24):
    results=[]
    for row in rows:
        generated=generate_reply(model,row,max_new_tokens=budget)
        results.append({'id':row['id'],'input':row['input'],'reference':row['response'],**generated,**response_checks(generated['text'],row['response']),'human_review':None})
    return results
# Small plotting helpers display measured tensors and counts.
def bar_plot(labels, values, title, ylabel):
    fig, ax = plt.subplots(figsize=(7, 3.5))
    ax.bar(labels, np.asarray(values))
    ax.set(title=title, ylabel=ylabel)
    if np.all(np.asarray(values)==0):
        ax.set_ylim(0,1)
        ax.text(0.5,0.5,'All values are zero',transform=ax.transAxes,ha='center')
    fig.tight_layout()
    return fig

def heat_plot(values, title, xlabel, ylabel, label):
    array = values.detach().cpu().numpy() if isinstance(values, torch.Tensor) else np.asarray(values)
    fig, ax = plt.subplots(figsize=(7, 3.5))
    if np.any(array < 0):
        bound = max(float(np.abs(array).max()), 1e-8)
        view = ax.imshow(array, aspect='auto', interpolation='nearest', cmap='coolwarm', vmin=-bound, vmax=bound)
    else:
        view = ax.imshow(array, aspect='auto', interpolation='nearest', cmap='viridis', vmin=0, vmax=max(float(array.max()),1e-8))
    ax.set_xticks(np.unique(np.linspace(0,array.shape[1]-1,min(8,array.shape[1])).astype(int)))
    ax.set_yticks(np.unique(np.linspace(0,array.shape[0]-1,min(8,array.shape[0])).astype(int)))
    ax.set(title=title, xlabel=xlabel, ylabel=ylabel)
    fig.colorbar(view, ax=ax, label=label)
    fig.tight_layout()
    return fig

Guided task 01

Write the evaluation contract before training

A final project needs a question that its evidence can answer. Here the question is narrow: how does a fixed amount of response-only training change this tiny model on the supplied item-packing requests? It is not whether the model has become a general assistant. The input format, character vocabulary, update count, greedy response budget and data roles are fixed before inspecting final test outcomes.

Use validation examples to understand the workflow and its limitations. Keep the final test situations for a single paired report after the procedure is settled. Baseline and candidate must receive the same prompts under the same decoding policy. Record both teacher-forced response loss and freely generated answers; neither substitutes for the other. Automated checks will include normalized exact match and a deliberately limited output-shape rule. A human rubric will address whether the item actually suits the situation. Defining these meanings first makes it harder to silently replace an awkward result with a more favorable metric after seeing it.

  3. Count the examples assigned to each role.

Guided task 02

Build a controlled baseline and adaptation

The project repeats a short training run so it is independently executable after setup. Both baseline and candidate begin from the same randomly initialized tiny model. The candidate receives forty response-only updates, while the baseline is retained unchanged. The optional extension uses a real pretrained checkpoint instead, with its own tokenizer and a smaller selected parameter subset. Keep those origins explicit when comparing artifacts.

The update loop follows the declared budget without using final test outcomes. It records actual training losses but does not assert that any quality metric must improve. The response mask and true EOS target are identical to the previous labs, and parameter sharing remains intact. Training mode enables the intended dropout behavior; evaluation functions later use eval and no_grad. By retaining both parameter states, the final report can compare the same requests against the original baseline rather than a newly drawn random model. This controlled difference makes the observed changes attributable to this specific adaptation procedure within the limits of the tiny fixture.

Try it

  1. Clone one initialized baseline to create the candidate.
  2. Run exactly the declared forty training updates.
  3. Preserve the baseline and plot actual training objectives.
Python · run in order
baseline=TinyInstructionLM().eval()
candidate=copy.deepcopy(baseline)

optimizer=torch.optim.AdamW(candidate.parameters(),lr=project_config['learning_rate'],weight_decay=project_config['weight_decay'])

training_curve=[]

    candidate.train()
    optimizer.zero_grad(set_to_none=True)

    loss=total/count
    loss.backward()
    optimizer.step()
    training_curve.append(float(loss.detach()))

ax.plot(range(1,len(training_curve)+1),training_curve)
ax.set(xlabel='Optimizer update',ylabel='Mean response loss',title='Actual fixed-budget project training')
fig.tight_layout()

Guided task 03

Extract responses by token position

Generation APIs often return the prompt together with its continuation. The response boundary is the number of input token IDs, so extracting the continuation should use token positions. Replacing a prompt substring in decoded text is fragile: a generated answer may repeat part of the request legitimately, and tokenizer decoding can alter whitespace presentation. The local helper already returns the newly generated IDs separately.

Inspect an illustrative extraction first, then collect actual validation continuations from both models. Decode only newly generated character IDs and omit the EOS marker from visible text. Keep the raw generated IDs and stopping reason as evidence, including an immediate EOS or a budget-limited repeated sequence. A response with no visible text is a real outcome rather than a missing display value. The same principle applies in the GPT-2 extension, where the returned tensor is sliced after the prompt width before decoding. This separation protects the evaluation from accidentally counting the supplied instruction as a successful generated answer.

Guided task 04

Apply narrow automated checks honestly

Exact match compares the entire normalized response with one supplied reference. The normalization here lowercases text and collapses whitespace, so capitalization and repeated spaces do not create superficial mismatches. It does not erase extra words or replace synonyms. A useful answer with different wording can fail exact match; a memorized reference can pass without proving broad understanding.

The format rule is intentionally limited: the response must contain two to four ASCII alphabetic words and begin with bring. It checks a simple expected shape, not whether the chosen item is sensible, safe or the only item named. Keep this rule distinct from semantic judgment. For example, bring a pillow has the expected shape but may not match a particular reference. Test these rule boundaries on explicit strings before applying them to actual generated responses. The resulting flags are reproducible and easy to audit, which makes them valuable components of a report as long as their restricted meaning stays visible.

Guided task 05

Define a human rubric without inventing scores

Some response qualities require interpretation of the situation. The review rubric asks whether the item is useful for the stated visit or weather, whether the response follows the one-item request, and whether it avoids unrelated additions. A reviewer should inspect the input and actual response together, then record a judgment and a short reason under each criterion. The supplied reference is an example answer, not the whole space of acceptable advice.

This worksheet leaves human-review fields unscored. Missing review is represented by None and displayed as pending, never as zero and never as an invented favorable rating. A zero would mean a reviewer applied a scale and found failure; pending means that measurement has not happened. More extensive projects can use multiple reviewers and discuss disagreement. A model-based judge is an optional source of additional opinion, not objective truth, and would require its own prompt, version and failure analysis. No paid judge or external service is necessary for this project. Keep automated flags alongside the rubric so readers know which evidence was computed and which still needs human judgment.

Guided task 06

Run the fixed final paired test

Now evaluate the untouched final test situations under the already declared settings. Both models receive the same prompt-only inputs, decoding budget and task checks. Their teacher-forced losses use the same reference targets and token denominator. No additional update or configuration selection follows this measurement. Report all intended test records rather than retaining only a favorable example.

Summaries should preserve their units. Response loss is a mean over supervised tokens; exact-match, format and EOS-stop rates are fractions of examples. Plotting them on separate axes prevents a numeric loss from looking like a success percentage. The EOS rate measures learned termination under the generation policy, not semantic correctness. With only two final examples, a single changed outcome moves an example fraction by one half. This coarse resolution is a reason to keep the individual records visible and describe the study as a mechanics demonstration. A larger evaluation would require a more representative collection and a predeclared sampling and uncertainty analysis.

Guided task 07

Export failure details and provenance

A useful report preserves individual outputs and the circumstances that produced them. Store record IDs, situations, references, generated text, generated IDs, stopping reasons and automated flags for both models. Keep human review explicitly absent until it occurs. A summary count alone cannot explain whether failure came from empty output, repetition, an alternate reasonable item or a malformed reply.

Save the report in a fresh temporary directory with model configuration, vocabulary, special tokens, formatter, data IDs and normalization rules. Record the library version and randomly initialized provenance. This makes the experiment interpretable without relying on an unstated notebook variable. The error-count plot uses overlapping categories: one response may fail exact match, fail format and reach the budget simultaneously. Therefore these bars must not be added as though they count distinct failed examples. The report is a transparent record of this small run, not a deployment certificate. Preserve failures even when they make the limitations of the model obvious.

Guided task 08

Verify the final model handoff

Save the adapted model alongside the report, including its architecture and input contract. Optimizer state, completed steps and RNG state make the training history explicit, while a fresh-model reload checks the inference handoff. The receiving code must construct the same vocabulary size, feature width, attention heads and context capacity, with tied embeddings restored by the constructor.

Compare numeric logits on the same final inputs, then regenerate responses under the recorded greedy budget. Equal decoded strings alone would be a weaker check because different token paths or logits can sometimes produce the same visible text. Retain the token IDs and assert their replay as well. This verifies persistence of this run in the current environment, not universal reproducibility across every future library release. The complete project now contains a baseline comparison, separate automated and pending human evidence, explicit limitations and a tested artifact. Extension 1

Optional real pretrained instruction evaluation project

This standalone route loads the genuine GPT-2 checkpoint with its own subword tokenizer and unchanged architecture. Expect about 500 MB of pretrained weights, at least 8 GB of host RAM, and a longer CPU run than the tiny core; a GPU is preferable for larger experiments. Four short updates adapt only the final transformer block and final layer normalization. Earlier layers and the tied input/output embeddings remain frozen. The measured before/after values demonstrate mechanics on constructed held-out situations; they do not establish a useful general assistant. Prompt and space-prefixed response are encoded separately to preserve an explicit target boundary, and the complete response plus EOS must fit. The library shifts the full labels internally. The saved adaptation contains parameter overrides and the exact base-checkpoint identity; reconstruct that base before applying this overlay. Human-review fields remain unscored until a person applies the rubric. The final replay constructs fresh model and tokenizer objects from the recorded revision, applies the saved parameter overlay, and compares the generated token IDs.

Python · standalone extension
import json, tempfile
from pathlib import Path
import torch
from transformers import GPT2TokenizerFast,GPT2LMHeadModel
torch.set_num_threads(1)
torch.manual_seed(219)
checkpoint='openai-community/gpt2'
revision='607a30d783dfa663caf39e06633721c8d4cfcd7e'
tokenizer=GPT2TokenizerFast.from_pretrained(checkpoint,revision=revision)
tokenizer.pad_token=tokenizer.eos_token
tokenizer.padding_side='right'
model=GPT2LMHeadModel.from_pretrained(checkpoint,revision=revision,use_safetensors=True)
model.config.pad_token_id=tokenizer.eos_token_id
training=[('library','bring a book'),('park','bring water'),('rain','bring an umbrella'),('gym','bring a towel')]
validation=[('reading cafe','bring a book'),('workout','bring a towel')]
test=[('hot day','bring water'),('wet weather','bring an umbrella')]

def prompt(situation):
    return 'Task: name one useful item\nInput: '+situation+'\nReply:'

def batch(rows):
    segments=[]
    for situation,response in rows:
        if not response.strip():
            raise ValueError('Response must be nonempty.')
        prefix=tokenizer.encode(prompt(situation),add_special_tokens=False)
        answer=tokenizer.encode(' '+response.strip(),add_special_tokens=False)+[tokenizer.eos_token_id]
        if len(prefix)+len(answer)>96:
            raise ValueError('Complete reference and EOS must fit.')
        segments.append((prefix,answer))
    width=max(len(p)+len(a) for p,a in segments)
    ids=torch.full((len(rows),width),tokenizer.eos_token_id,dtype=torch.long)
    mask=torch.zeros_like(ids)
    labels=torch.full_like(ids,-100)
    for i,(prefix,answer) in enumerate(segments):
        full=prefix+answer
        ids[i,:len(full)]=torch.tensor(full)
        mask[i,:len(full)]=1
        labels[i,len(prefix):len(full)]=torch.tensor(answer)
    return {'input_ids':ids,'attention_mask':mask,'labels':labels}

def evaluate(rows):
    model.eval()
    total,count=0.0,0
    with torch.no_grad():
        for row in rows:
            data=batch([row])
            n=int((data['labels'][:,1:]!=-100).sum())
            # GPT2LMHeadModel shifts full labels internally: do not shift here.
            loss=model(**data,use_cache=False).loss
            total+=float(loss)*n
            count+=n
    return {'loss':total/count,'tokens':count,'records':len(rows)}

def replies(rows):
    model.eval()
    outputs=[]
    with torch.no_grad():
        for situation,reference in rows:
            data=tokenizer(prompt(situation),return_tensors='pt',add_special_tokens=False)
            # Generate one unpadded prompt at a time; retain its attention mask.
            generated=model.generate(**data,max_new_tokens=20,do_sample=False,pad_token_id=tokenizer.eos_token_id,eos_token_id=tokenizer.eos_token_id)
            new_ids=generated[0,data['input_ids'].shape[1]:]
            outputs.append({'input':situation,'reference':reference,'response':tokenizer.decode(new_ids,skip_special_tokens=True).strip(),'new_token_ids':new_ids.tolist()})
    return outputs

for parameter in model.parameters():
    parameter.requires_grad=False
for module in [model.transformer.h[-1],model.transformer.ln_f]:
    for parameter in module.parameters():
        parameter.requires_grad=True
trainable_names=[name for name,p in model.named_parameters() if p.requires_grad]
initial_overrides={name:p.detach().clone() for name,p in model.named_parameters() if p.requires_grad}
before=evaluate(validation)
before_replies=replies(validation)
optimizer=torch.optim.AdamW([p for p in model.parameters() if p.requires_grad],lr=0.00005)
training_batch=batch(training)
for step in range(4):
    model.train()
    optimizer.zero_grad(set_to_none=True)
    loss=model(**training_batch,use_cache=False).loss
    loss.backward()
    optimizer.step()
after=evaluate(validation)
after_replies=replies(validation)
print({'before':before,'after':after,'trainable_parameters':sum(p.numel() for p in model.parameters() if p.requires_grad),'steps':4})
print({'before_responses':before_replies,'after_responses':after_replies})
assert before['tokens']==after['tokens'] and torch.isfinite(loss)
assert any(not torch.equal(initial_overrides[name],parameter) for name,parameter in model.named_parameters() if parameter.requires_grad)

# All settings above are fixed before this final paired test comparison.
adapted_overrides={name:p.detach().clone() for name,p in model.named_parameters() if p.requires_grad}
adapted_test=evaluate(test)
adapted_responses=replies(test)
model.load_state_dict(initial_overrides,strict=False)
baseline_test=evaluate(test)
baseline_responses=replies(test)
model.load_state_dict(adapted_overrides,strict=False)
normalize=lambda text:' '.join(text.lower().split())
for before_row,after_row in zip(baseline_responses,adapted_responses):
    before_row['exact_match']=normalize(before_row['response'])==normalize(before_row['reference'])
    after_row['exact_match']=normalize(after_row['response'])==normalize(after_row['reference'])
    before_row['human_review']=None
    after_row['human_review']=None
report={'base_checkpoint':checkpoint,'revision':revision,'steps':4,'trainable_names':trainable_names,'format':'Task: name one useful item\nInput: {situation}\nReply:','segment_encoding':'Encode prompt and space-prefixed answer separately; append true EOS.','response_budget':20,'decoding':'greedy','baseline_test':baseline_test,'adapted_test':adapted_test,'baseline_responses':baseline_responses,'adapted_responses':adapted_responses,'human_rubric':['Useful item for the situation','Follows the one-item request','No unrelated extra content'],'interpretation':'Tiny mechanics study; exact match is not a complete semantic assessment.'}
folder=Path(tempfile.mkdtemp(prefix='gpt2_instruction_project_'))
(folder/'report.json').write_text(json.dumps(report,indent=2),encoding='utf8')
torch.save({'overrides':adapted_overrides,'base_checkpoint':checkpoint,'revision':revision,'trainable_names':trainable_names,'optimizer':optimizer.state_dict(),'steps':4,'rng_state':torch.get_rng_state(),'tokenizer_checkpoint':checkpoint,'tokenizer_revision':revision,'prompt_format':report['format'],'segment_encoding':report['segment_encoding'],'response_budget':20,'decoding':'greedy'},folder/'adaptation.pt')
loaded=torch.load(folder/'adaptation.pt',weights_only=True)
# This compact artifact is an overlay, not a standalone complete model.
assert set(loaded['overrides'])==set(loaded['trainable_names'])
assert loaded['prompt_format']==report['format'] and loaded['response_budget']==20
assert loaded['decoding']=='greedy'
# Reconstruct both objects from the recorded base identities before applying the overlay.
del model
model=GPT2LMHeadModel.from_pretrained(loaded['base_checkpoint'],revision=loaded['revision'],use_safetensors=True)
tokenizer=GPT2TokenizerFast.from_pretrained(loaded['tokenizer_checkpoint'],revision=loaded['tokenizer_revision'])
tokenizer.pad_token=tokenizer.eos_token
tokenizer.padding_side='right'
model.config.pad_token_id=tokenizer.eos_token_id
result=model.load_state_dict(loaded['overrides'],strict=False)
assert not result.unexpected_keys
assert set(result.missing_keys)==set(model.state_dict())-set(loaded['overrides'])
replayed=replies(test)
assert [r['new_token_ids'] for r in replayed]==[r['new_token_ids'] for r in adapted_responses]
print({'artifact_folder':str(folder),'paired_final_test':report})
