def draw_item(label, row_shift, col_shift, tint):
    image = np.full((16, 16, 3), 0.94, dtype=np.float32)
    if label == 0:
        image[5:13, 3:10] = [0.15, 0.4, tint]
        image[6:11, 10:14] = [0.15, 0.4, tint]
        image[7:10, 10:12] = 0.94
    elif label == 1:
        image[3:14, 3:13] = [tint, 0.22, 0.2]
        image[4:13, 5:12] = [0.88, 0.82, 0.62]
    else:
        image[2:5, 7:10] = [0.25, 0.35, 0.25]
        image[5:14, 5:12] = [0.2, tint, 0.32]
    return np.roll(image, (row_shift, col_shift), axis=(0, 1))
pictures, targets = [], []
for label in range(3):
    for sample in range(60):
        image = draw_item(label, int(rng.integers(-1,2)), int(rng.integers(-1,2)), float(rng.uniform(.55,.85)))
        image = np.clip(image + rng.normal(0, .015, image.shape), 0, 1).astype(np.float32)
        pictures.append(image)
        targets.append(label)
images = np.stack(pictures)
labels = np.array(targets, dtype=np.int32)
train_ids, val_ids, test_ids = [], [], []
for label in range(3):
    ids = rng.permutation(np.flatnonzero(labels == label))

train_ids, val_ids, test_ids = [rng.permutation(ids) for ids in (train_ids,val_ids,test_ids)]

def batches(x, y, batch_size=16, shuffle=False):
    ds = tf.data.Dataset.from_tensor_slices((x, y))
    if shuffle:
        ds = ds.shuffle(len(x), seed=123, reshuffle_each_iteration=True)
    options = tf.data.Options()
    options.threading.private_threadpool_size = 1
    options.threading.max_intra_op_parallelism = 1
    return ds.batch(batch_size).with_options(options)
