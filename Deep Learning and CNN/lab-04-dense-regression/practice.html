# Invented planning records; these are not appliance measurements.
n = 180
mass_kg = rng.uniform(1.0, 7.0, n)
item_count = rng.integers(4, 25, n)
features = np.column_stack([mass_kg, item_count]).astype(np.float32)
minutes = (8 + 2.5*mass_kg + 0.35*item_count + rng.normal(0, 0.8, n)).astype(np.float32)[:, None]
record_ids = rng.permutation(n)
train_ids, val_ids, test_ids = record_ids[:108], record_ids[108:144], record_ids[144:]
x_train, y_train = features[train_ids], minutes[train_ids]
x_val, y_val = features[val_ids], minutes[val_ids]
x_test, y_test = features[test_ids], minutes[test_ids]
feature_names = ["Mass (kg)", "Item count"]

def batches(x, y, batch_size=16, shuffle=False):
    ds = tf.data.Dataset.from_tensor_slices((x, y))
    if shuffle:
        ds = ds.shuffle(len(x), seed=123, reshuffle_each_iteration=True)
    options = tf.data.Options()
    options.threading.private_threadpool_size = 1
    options.threading.max_intra_op_parallelism = 1
    return ds.batch(batch_size).with_options(options)
