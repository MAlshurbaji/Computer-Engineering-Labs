import numpy as np
import tensorflow as tf
import keras
import matplotlib.pyplot as plt
keras.utils.set_random_seed(114)
rng = np.random.default_rng(114)

def batches(x, y, batch_size=16, shuffle=False):
    ds = tf.data.Dataset.from_tensor_slices((x, y))
    if shuffle:
        ds = ds.shuffle(len(x), seed=123, reshuffle_each_iteration=True)
    options = tf.data.Options()
    options.threading.private_threadpool_size = 1
    options.threading.max_intra_op_parallelism = 1
    return ds.batch(batch_size).with_options(options)

# Draw paper notes on dark desk backgrounds. Every seventh desk is empty.
def desks(ids):
    images, masks = [], []
    for ident in ids:
        local = np.random.default_rng(900 + int(ident))
        mask = np.zeros((16,16,1),dtype="float32")
        if ident % 7 != 0:
            top, left = local.integers(2,7,size=2)
            height, width = local.integers(5,9,size=2)
            mask[top:top+height,left:left+width,0] = 1
        background = np.full((16,16,3),0.10,dtype="float32")
        color = np.array([0.75,0.80,0.45],dtype="float32")
        image = background*(1-mask) + color*mask
        image = np.clip(image + local.normal(0,0.035,image.shape),0,1)
        images.append(image.astype("float32"))
        masks.append(mask)
    return np.stack(images),np.stack(masks)

train_ids = np.arange(0,64)

test_ids = np.arange(80,96)
x_train, m_train = desks(train_ids)
x_val, m_val = desks(val_ids)
x_test, m_test = desks(test_ids)

def segmenter():
    model = keras.Sequential([
        keras.Input((16,16,3)),
        keras.layers.Conv2D(4,3,padding="same",activation="relu"),
        keras.layers.Conv2D(4,3,padding="same",activation="relu"),
        keras.layers.Conv2D(1,1,activation="sigmoid")
    ])
    model.compile(optimizer=keras.optimizers.Adam(0.005),
                  loss=keras.losses.BinaryCrossentropy())
    return model

# Per-image foreground IoU; an empty prediction with empty truth scores one.
def image_iou(truth, prediction):
    truth, prediction = np.asarray(truth,dtype=bool), np.asarray(prediction,dtype=bool)
    axes = tuple(range(1,truth.ndim))
    intersection = np.sum(truth & prediction,axis=axes)
    union = np.sum(truth | prediction,axis=axes)
    return np.divide(intersection,union,out=np.ones_like(union,dtype=float),where=union!=0)
